## 2. Теоретическая справка

### 2.1. Инварианты класса

**Инвариант класса** — это логическое условие, которое должно быть истинно для всех корректных состояний объекта в любой момент времени между вызовами его методов.

**Примеры инвариантов:**
- `balance >= 0` (баланс не может быть отрицательным)
- `age >= 0 && age <= 150` (возраст в допустимом диапазоне)
- `totalPrice == sum(item.price * item.quantity)` (цена равна сумме компонентов)

### 2.2. Техника "Состояние-Действие-Состояние"

**Структура теста:**
1. **State:** Проверка начального состояния объекта.
2. **Action:** Вызов тестируемого метода.
3. **State:** Проверка конечного состояния объекта.

### 2.3. Проверка состояния

При проверке состояния необходимо:
1. Проверить все поля объекта, влияющие на поведение.
2. Проверить инварианты до и после действия.
3. Проверить состояние как при успешных, так и при ошибочных сценариях.

public class TaskQueue {
    private List<Task> tasks;
    private int maxSize;
    private Task currentTask;
    private boolean isProcessing;
    private int completedCount;
    private int failedCount;
    
    public TaskQueue(int maxSize) {
        this.maxSize = maxSize;
        this.tasks = new ArrayList<>();
        this.isProcessing = false;
        this.completedCount = 0;
        this.failedCount = 0;
    }
    
    public void add(Task task) {
        if (tasks.size() >= maxSize) {
            throw new IllegalStateException("Queue is full");
        }
        if (task == null) {
            throw new IllegalArgumentException("Task cannot be null");
        }
        tasks.add(task);
    }
    
    public Task take() {
        if (tasks.isEmpty()) {
            throw new IllegalStateException("Queue is empty");
        }
        currentTask = tasks.remove(0);
        isProcessing = true;
        return currentTask;
    }
    
    public void complete() {
        if (!isProcessing || currentTask == null) {
            throw new IllegalStateException("No task in progress");
        }
        completedCount++;
        currentTask = null;
        isProcessing = false;
    }
    
    public void fail() {
        if (!isProcessing || currentTask == null) {
            throw new IllegalStateException("No task in progress");
        }
        failedCount++;
        currentTask = null;
        isProcessing = false;
    }
    
    public void clear() {
        if (isProcessing) {
            throw new IllegalStateException("Cannot clear while processing");
        }
        tasks.clear();
    }
}
Инварианты:

Если isProcessing == true, то currentTask != null

    private LightState previousState;
    private int timer;
    private int redDuration;
    private int greenDuration;
    private int yellowDuration;
    private boolean isPedestrianMode;
    private List<LightState> history;
    
    public TrafficLight(int redDuration, int greenDuration, int yellowDuration) {
        this.currentState = LightState.RED;
        this.previousState = LightState.RED;
        this.redDuration = redDuration;
        this.greenDuration = greenDuration;
        this.yellowDuration = yellowDuration;
        this.timer = redDuration;
        this.isPedestrianMode = false;
        this.history = new ArrayList<>();
        this.history.add(LightState.RED);
    }
    
    public void tick() {
        timer--;
        if (timer <= 0) {
            switch (currentState) {
                case RED:
                    changeState(LightState.GREEN, greenDuration);
                    break;
                case GREEN:
                    changeState(LightState.YELLOW, yellowDuration);
                    break;
                case YELLOW:
                    changeState(isPedestrianMode ? LightState.RED : LightState.GREEN, 
                                isPedestrianMode ? redDuration : greenDuration);
                    isPedestrianMode = false;
                    break;
            }
        }
    }
    
    private void changeState(LightState newState, int duration) {
        previousState = currentState;
        currentState = newState;
        timer = duration;
        history.add(newState);
    }
    
    public void activatePedestrianMode() {
        if (currentState != LightState.GREEN) {
            throw new IllegalStateException("Pedestrian mode only available on GREEN");
        }
        isPedestrianMode = true;
    }
}

public class Schedule {
    private List<Event> events;
    private LocalDate startDate;
    private LocalDate endDate;
    private int capacity;
    private int currentCount;
    private boolean isFull;
    
    public Schedule(LocalDate startDate, LocalDate endDate, int capacity) {
        this.startDate = startDate;
        this.endDate = endDate;
        this.capacity = capacity;
        this.events = new ArrayList<>();
        this.currentCount = 0;
        this.isFull = false;
    }
    
    public void addEvent(Event event) {
        if (isFull) {
            throw new IllegalStateException("Schedule is full");
        }
        if (!event.getDate().isAfter(startDate) || !event.getDate().isBefore(endDate)) {
            throw new IllegalArgumentException("Event date out of schedule range");
        }
        if (hasConflict(event)) {
            throw new IllegalStateException("Event conflicts with existing event");
        }
        events.add(event);
        currentCount++;
        if (currentCount >= capacity) {
            isFull = true;
        }
        Collections.sort(events);
    }
    
    public void removeEvent(Event event) {
        events.remove(event);
        currentCount--;
        isFull = false;

    public List<Event> getEventsForDate(LocalDate date) {
        return events.stream()
            .filter(e -> e.getDate().equals(date))
            .collect(Collectors.toList());
    }
    
    private boolean hasConflict(Event event) {
        return events.stream().anyMatch(e -> e.overlaps(event));
    }
}

public class Stack<T> {
    private List<T> elements;
    private int maxSize;
    private T topElement;
    private int size;
    
    public Stack(int maxSize) {
        this.maxSize = maxSize;
        this.elements = new ArrayList<>();
        this.size = 0;
        this.topElement = null;
    }
    
    public void push(T element) {
        if (element == null) {
            throw new IllegalArgumentException("Element cannot be null");
        }
        if (size >= maxSize) {
            throw new IllegalStateException("Stack is full");
        }
        elements.add(element);
        topElement = element;
        size++;
    }
    
    public T pop() {
        if (isEmpty()) {
            throw new IllegalStateException("Stack is empty");
        }
        T element = elements.remove(size - 1);
        size--;
        topElement = isEmpty() ? null : elements.get(size - 1);
        return element;
    }
    
    public T peek() {
        if (isEmpty()) {
            throw new IllegalStateException("Stack is empty");
        }
        return topElement;
    }
    
    public boolean isEmpty() {
        return size == 0;
    }
    
    public boolean isFull() {
        return size == maxSize;
    }
    
    public void clear() {
        elements.clear();
        size = 0;
        topElement = null;
    }
}

            throw new IllegalStateException("Counter would exceed maximum");

        value = newValue;

            throw new IllegalStateException("Counter is paused");

            throw new IllegalArgumentException("Delta must be positive");

            throw new IllegalStateException("Counter would go below minimum");

        value = newValue;
        saveState();

        this.fee = fee;

        this.createdAt = LocalDateTime.now();
        this.events = new ArrayList<>();
        addEvent("CREATED");
    }
    
    public void process(BigDecimal fromBalance, BigDecimal toBalance) {
        if (status != TransactionStatus.PENDING) {
            throw new IllegalStateException("Only pending transactions can be processed");
        }
        if (fromBalance.compareTo(amount.add(fee)) < 0) {
            status = TransactionStatus.FAILED;
            addEvent("FAILED: Insufficient balance");
            return;
        }
        this.fromBalanceBefore = fromBalance;
        this.toBalanceBefore = toBalance;
        this.fromBalanceAfter = fromBalance.subtract(amount).subtract(fee);
        this.toBalanceAfter = toBalance.add(amount);
        status = TransactionStatus.COMPLETED;
        processedAt = LocalDateTime.now();
        addEvent("PROCESSED");
    }
    
    public void reverse() {
        if (status != TransactionStatus.COMPLETED) {
            throw new IllegalStateException("Only completed transactions can be reversed");
        }
        status = TransactionStatus.REVERSED;
        addEvent("REVERSED");
    }
    
    public void retry() {
        if (status != TransactionStatus.FAILED) {
            throw new IllegalStateException("Only failed transactions can be retried");
        }
        status = TransactionStatus.PENDING;
        addEvent("RETRY");
    }
}
Инварианты:

amount.compareTo(BigDecimal.ZERO) > 0

fee.compareTo(BigDecimal.ZERO) >= 0

fromAccount != null && toAccount != null

fromAccount != toAccount

Если status == COMPLETED, то поля баланса заполнены

Если status == PENDING, то processedAt == null

            throw new IllegalStateException("Session is no longer active");
        }

            ipAddresses.add(ip);
        }
        this.currentDevice = device;
    }
    
    public void enableTwoFactorAuth() {

            throw new IllegalStateException("2FA already enabled");
        }
        isTwoFactorAuthEnabled = true;
        isTwoFactorAuthVerified = false;
    }
    
    public void verifyTwoFactorAuth(String code) {
        if (!isTwoFactorAuthEnabled) {
            throw new IllegalStateException("2FA is not enabled");
        }
        if (isTwoFactorAuthVerified) {
            throw new IllegalStateException("2FA already verified");
        }
        if (validate2FACode(code)) {
            isTwoFactorAuthVerified = true;
        } else {
            failedLoginAttempts++;
            if (failedLoginAttempts >= 3) {
                status = SessionStatus.BLOCKED;
            }
            throw new AuthenticationException("Invalid 2FA code");
        }
    }

        if (status == SessionStatus.TERMINATED) {
            throw new IllegalStateException("Session already terminated");
        }
        status = SessionStatus.EXPIRED;
    }
    
    public void terminate() {
        if (status == SessionStatus.EXPIRED) {
            throw new IllegalStateException("Session already expired");
        }
        status = SessionStatus.TERMINATED;
    }
    
    public boolean isExpired() {
        if (status == SessionStatus.EXPIRED || status == SessionStatus.TERMINATED) {
            return true;
        }
        Duration duration = Duration.between(lastActivityTime, LocalDateTime.now());
        return duration.getSeconds() > sessionTimeoutSeconds;
    }
