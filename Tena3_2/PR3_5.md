public interface SortAlgorithm {
    <T extends Comparable<T>> void sort(List<T> list);
    String getName();
    long getComparisonCount();
    long getSwapCount();
    void resetStats();
}

public abstract class BaseSort implements SortAlgorithm {
    protected long comparisonCount;
    protected long swapCount;
    
    @Override
    public long getComparisonCount() { return comparisonCount; }
    
    @Override
    public long getSwapCount() { return swapCount; }
    
    @Override
    public void resetStats() {
        comparisonCount = 0;
        swapCount = 0;
    }
    
    protected <T extends Comparable<T>> boolean compare(T a, T b) {
        comparisonCount++;
        return a.compareTo(b) <= 0;
    }
    
    protected <T> void swap(List<T> list, int i, int j) {
        swapCount++;
        Collections.swap(list, i, j);
    }
}

public class BubbleSort extends BaseSort {
    @Override
    public <T extends Comparable<T>> void sort(List<T> list) {
        if (list == null || list.isEmpty()) return;
        resetStats();
        int n = list.size();
        for (int i = 0; i < n - 1; i++) {
            for (int j = 0; j < n - i - 1; j++) {
                if (!compare(list.get(j), list.get(j + 1))) {
                    swap(list, j, j + 1);
                }
            }
        }
    }
    
    @Override
    public String getName() { return "BUBBLE_SORT"; }
}

public class QuickSort extends BaseSort {
    @Override
    public <T extends Comparable<T>> void sort(List<T> list) {
        if (list == null || list.isEmpty()) return;
        resetStats();
        quickSort(list, 0, list.size() - 1);
    }
    
    private <T extends Comparable<T>> void quickSort(List<T> list, int low, int high) {
        if (low < high) {
            int pi = partition(list, low, high);
            quickSort(list, low, pi - 1);
            quickSort(list, pi + 1, high);
        }
    }
    
    private <T extends Comparable<T>> int partition(List<T> list, int low, int high) {
        T pivot = list.get(high);
        int i = low - 1;
        for (int j = low; j < high; j++) {
            if (compare(list.get(j), pivot)) {
                i++;
                swap(list, i, j);
            }
        }
        swap(list, i + 1, high);
        return i + 1;
    }
    
    @Override
    public String getName() { return "QUICK_SORT"; }
}

public class MergeSort extends BaseSort {
    @Override
    public <T extends Comparable<T>> void sort(List<T> list) {
        if (list == null || list.isEmpty()) return;
        resetStats();
        mergeSort(list, 0, list.size() - 1);
    }
    
    private <T extends Comparable<T>> void mergeSort(List<T> list, int left, int right) {
        if (left < right) {
            int mid = (left + right) / 2;
            mergeSort(list, left, mid);
            mergeSort(list, mid + 1, right);
            merge(list, left, mid, right);
        }
    }
    
    private <T extends Comparable<T>> void merge(List<T> list, int left, int mid, int right) {
        // Реализация слияния
        List<T> temp = new ArrayList<>(list.subList(left, right + 1));
        int i = 0, j = mid - left + 1, k = left;
        while (i <= mid - left && j < right - left + 1) {
            if (compare(temp.get(i), temp.get(j))) {
                list.set(k++, temp.get(i++));
            } else {
                list.set(k++, temp.get(j++));
            }
        }
        // Остаток...
    }
    
    @Override
    public String getName() { return "MERGE_SORT"; }
}

public interface Encryption {
    String encrypt(String data);
    String decrypt(String data);
    String getAlgorithm();
    int getKeySize();
}

public abstract class BaseEncryption implements Encryption {
    protected String algorithm;
    protected int keySize;
    
    @Override
    public String getAlgorithm() { return algorithm; }
    
    @Override
    public int getKeySize() { return keySize; }
}

public class AESEncryption extends BaseEncryption {
    private String key;
    
    public AESEncryption(String key) {
        this.algorithm = "AES";
        this.keySize = 256;
        this.key = key;
    }
    
    @Override
    public String encrypt(String data) {
        if (data == null || data.isEmpty()) {
            throw new IllegalArgumentException("Data cannot be empty");
        }
        if (key.length() < 32) {
            throw new IllegalStateException("Key must be at least 32 characters");
        }
        // Имитация шифрования AES
        return "AES:" + data;
    }
    
    @Override
    public String decrypt(String data) {
        if (data == null || !data.startsWith("AES:")) {
            throw new IllegalArgumentException("Invalid encrypted data");
        }
        return data.substring(4);
    }
}

public class RSAEncryption extends BaseEncryption {
    private String publicKey;
    private String privateKey;
    
    public RSAEncryption(String publicKey, String privateKey) {
        this.algorithm = "RSA";
        this.keySize = 2048;
        this.publicKey = publicKey;
        this.privateKey = privateKey;
    }
    
    @Override
    public String encrypt(String data) {
        if (data == null || data.isEmpty()) {
            throw new IllegalArgumentException("Data cannot be empty");
        }
        if (publicKey == null) {
            throw new IllegalStateException("Public key not set");
        }
        return "RSA:" + data;
    }
    
    @Override
    public String decrypt(String data) {
        if (data == null || !data.startsWith("RSA:")) {
            throw new IllegalArgumentException("Invalid encrypted data");
        }
        if (privateKey == null) {
            throw new IllegalStateException("Private key not set");
        }
        return data.substring(4);
    }
}

public class CaesarEncryption extends BaseEncryption {
    private int shift;
    
    public CaesarEncryption(int shift) {
        this.algorithm = "CAESAR";
        this.keySize = 0;
        this.shift = shift;
    }
    
    @Override
    public String encrypt(String data) {
        if (data == null || data.isEmpty()) {
            throw new IllegalArgumentException("Data cannot be empty");
        }
        StringBuilder result = new StringBuilder();
        for (char c : data.toCharArray()) {
            if (Character.isLetter(c)) {
                char base = Character.isUpperCase(c) ? 'A' : 'a';
                result.append((char) (base + (c - base + shift) % 26));
            } else {
                result.append(c);
            }
        }
        return result.toString();
    }
    
    @Override
    public String decrypt(String data) {
        return encrypt(data); // Для Цезаря шифрование и дешифрование одинаковы
    }
}

public interface Storage<T> {
    void save(String key, T data);
    T load(String key);
    boolean exists(String key);
    void delete(String key);
    List<String> listKeys();
    long getSize();
}

public abstract class BaseStorage<T> implements Storage<T> {
    protected Map<String, T> storage;
    protected long size;
    
    public BaseStorage() {
        this.storage = new HashMap<>();
        this.size = 0;
    }
    
    @Override
    public boolean exists(String key) {
        return storage.containsKey(key);
    }
    
    @Override
    public void delete(String key) {
        if (storage.remove(key) != null) {
            size--;
        }
    }
    
    @Override
    public List<String> listKeys() {
        return new ArrayList<>(storage.keySet());
    }
    
    @Override
    public long getSize() { return size; }
}

public class MemoryStorage<T> extends BaseStorage<T> {
    @Override
    public void save(String key, T data) {
        if (key == null || key.isEmpty()) {
            throw new IllegalArgumentException("Key cannot be null or empty");
        }
        if (!storage.containsKey(key)) {
            size++;
        }
        storage.put(key, data);
    }
    
    @Override
    public T load(String key) {
        if (!exists(key)) {
            throw new IllegalArgumentException("Key not found: " + key);
        }
        return storage.get(key);
    }
}

public class FileStorage<T> extends BaseStorage<T> {
    private String basePath;
    private Serializer<T> serializer;
    
    public FileStorage(String basePath, Serializer<T> serializer) {
        this.basePath = basePath;
        this.serializer = serializer;
    }
    
    @Override
    public void save(String key, T data) {
        if (key == null || key.isEmpty()) {
            throw new IllegalArgumentException("Key cannot be null or empty");
        }
        // Имитация сохранения в файл
        String path = basePath + "/" + key + ".dat";
        if (!exists(key)) {
            size++;
        }
        storage.put(key, data);
    }
    
    @Override
    public T load(String key) {
        if (!exists(key)) {
            throw new IllegalArgumentException("Key not found: " + key);
        }
        return storage.get(key);
    }
}

public class CacheStorage<T> extends BaseStorage<T> {
    private int maxSize;
    private Map<String, Long> accessTime;
    
    public CacheStorage(int maxSize) {
        this.maxSize = maxSize;
        this.accessTime = new HashMap<>();
    }
    
    @Override
    public void save(String key, T data) {
        if (storage.size() >= maxSize && !storage.containsKey(key)) {
            evict();
        }
        if (!storage.containsKey(key)) {
            size++;
        }
        storage.put(key, data);
        accessTime.put(key, System.currentTimeMillis());
    }
    
    @Override
    public T load(String key) {
        if (!exists(key)) {
            throw new IllegalArgumentException("Key not found: " + key);
        }
        accessTime.put(key, System.currentTimeMillis());
        return storage.get(key);
    }
    
    private void evict() {
        // Удаление наименее используемого элемента
        String oldest = accessTime.entrySet().stream()
            .min(Map.Entry.comparingByValue())
            .map(Map.Entry::getKey)
            .orElse(null);
        if (oldest != null) {
            storage.remove(oldest);
            accessTime.remove(oldest);
            size--;
        }
    }
}

public class RBACAuthorizer extends BaseAuthorizer {
    private Map<String, List<Permission>> rolePermissions;

        this.name = "RBAC";
        this.rolePermissions = new HashMap<>();
        // Инициализация ролей и прав
        rolePermissions.put("ADMIN", Arrays.asList(Permission.values()));
        rolePermissions.put("USER", Arrays.asList(Permission.READ, Permission.WRITE));
        rolePermissions.put("GUEST", Arrays.asList(Permission.READ));
    }
    
    @Override
    public boolean authorize(User user, Resource resource, Action action) {
        if (user == null || resource == null || action == null) {
            throw new IllegalArgumentException("Parameters cannot be null");
        }
        List<Permission> permissions = rolePermissions.getOrDefault(
            user.getRole(), new ArrayList<>()
        );
        Permission required = Permission.fromResourceAction(resource, action);
        return permissions.contains(required);
    }
    
    @Override
    public List<Permission> getPermissions(User user) {
        return rolePermissions.getOrDefault(user.getRole(), new ArrayList<>());
    }
}

    public NotNullValidator() {
        rules.add("NOT_NULL");
        errorMessage = "Value cannot be null";
    }

    public ValidationResult validate(T value) {
        if (value == null) {
            return fail(errorMessage);
        }
        return pass();
    }
    
    @Override
    public Validator<T> combine(Validator<T> other) {
        return new CompositeValidator<>(this, other);
    }
}

public class LengthValidator extends BaseValidator<String> {
    private int min;
    private int max;
    
    public LengthValidator(int min, int max) {
        this.min = min;
        this.max = max;
        this.rules.add("LENGTH");
        this.errorMessage = "Length must be between " + min + " and " + max;
    }
    
    @Override
    public ValidationResult validate(String value) {
        if (value == null) {
            return pass(); // Другой валидатор проверяет null
        }
        if (value.length() < min || value.length() > max) {
            return fail(errorMessage);
        }
        return pass();
    }
    
    @Override
    public Validator<String> combine(Validator<String> other) {
        return new CompositeValidator<>(this, other);
    }
}

public class EmailValidator extends BaseValidator<String> {
    private String pattern;
    
    public EmailValidator() {
        this.pattern = "^[A-Za-z0-9+_.-]+@(.+)$";
        this.rules.add("EMAIL");
        this.errorMessage = "Invalid email format";
    }
    
    @Override
    public ValidationResult validate(String value) {
        if (value == null) {
            return pass();
        }
        if (!value.matches(pattern)) {
            return fail(errorMessage);
        }
        return pass();
    }

    public Validator<String> combine(Validator<String> other) {
        return new CompositeValidator<>(this, other);

public class CompositeValidator<T> extends BaseValidator<T> {
    private List<Validator<T>> validators;
    
    public CompositeValidator(Validator<T>... validators) {
        this.validators = Arrays.asList(validators);
        this.rules = validators.stream()
            .flatMap(v -> v.getRules().stream())
            .collect(Collectors.toList());
    }
    
    @Override
    public ValidationResult validate(T value) {
        for (Validator<T> validator : validators) {
            ValidationResult result = validator.validate(value);
            if (!result.isValid()) {
                return result;
            }
        }
        return pass();
    }
    
    @Override
    public Validator<T> combine(Validator<T> other) {
        return new CompositeValidator<>(
            (CompositeValidator<T>) this,
            other
        );
    }
}

public abstract class BaseRouter implements Router {
    protected String algorithm;
    protected List<Route> routes;
    
    public BaseRouter() {
        this.routes = new ArrayList<>();
    }
    
    @Override
    public List<Route> getRoutes() { return new ArrayList<>(routes); }
    
    @Override
    public void addRoute(Route route) {
        if (route == null) {
            throw new IllegalArgumentException("Route cannot be null");
        }
        routes.add(route);
    }
    
    @Override
    public String getAlgorithm() { return algorithm; }
}

public class ShortestPathRouter extends BaseRouter {
    public ShortestPathRouter() {
        this.algorithm = "SHORTEST_PATH";
    }
    
    @Override
    public Route selectRoute(Point start, Point end) {
        if (start == null || end == null) {
            throw new IllegalArgumentException("Start and end points cannot be null");
        }
        // Имитация выбора кратчайшего пути
        if (routes.isEmpty()) {
            throw new IllegalStateException("No routes available");
        }
        return routes.stream()
            .min(Comparator.comparingDouble(r -> r.getDistance(start, end)))
            .orElseThrow(() -> new IllegalStateException("No route found"));
    }
}

    public FastestPathRouter() {
        this.algorithm = "FASTEST_PATH";

            throw new IllegalArgumentException("Start and end points cannot be null");
        }

            throw new IllegalStateException("No routes available");
        }
        // Имитация выбора самого быстрого пути
        return routes.stream()
            .min(Comparator.comparingDouble(r -> r.getEstimatedTime(start, end)))

public class PaymentSystemIntegrationTest {
    
    @Test
    void testPaymentSystem_CompleteWorkflow() {

        Map<String, Map<String, String>> params = Map.of(

            "PAYPAL", Map.of("email", "john@example.com")
        );
        
        List<Payment> payments = new ArrayList<>();
        payments.add(PaymentFactory.createPayment("CREDIT_CARD", params.get("CREDIT_CARD")));
        payments.add(PaymentFactory.createPayment("PAYPAL", params.get("PAYPAL")));
        
        // 2. Обработка всех платежей полиморфно
        double totalAmount = 0;
        for (Payment payment : payments) {
            payment.process(100.0);
            totalAmount += payment.getFee();
        }
        
        // 3. Проверка результатов
        assertTrue(payments.stream().allMatch(Payment::isCompleted));
        assertEquals(6.0, totalAmount, 0.001); // 2.5 + 3.5
    }
