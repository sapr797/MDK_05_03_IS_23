## 2. Теоретическая справка

### 2.1. Инструменты для мокирования файловой системы

| Инструмент | Назначение | Когда использовать |
|------------|------------|-------------------|
| `tmp_path` | Создание временной директории | Для тестов, работающих с реальной ФС |
| `monkeypatch` | Подмена функций и атрибутов | Для простых подмен (open, os.path.exists) |
| `pytest-mock` | Мокирование через `mocker.patch` | Для продвинутого мокирования |
| `pyfakefs` | Полная виртуальная файловая система | Для полной изоляции от реальной ФС |

### 2.2. Основные ошибки файловой системы

| Исключение | Когда возникает |
|------------|-----------------|
| `FileNotFoundError` | Файл или директория не существует |
| `PermissionError` | Недостаточно прав для чтения/записи/удаления |
| `IsADirectoryError` | Ожидался файл, но это директория |
| `NotADirectoryError` | Ожидалась директория, но это файл |
| `OSError` | Общая ошибка ОС (заполненный диск, и др.) |
| `FileExistsError` | Файл уже существует при попытке создания |

# log_handler.py
import os
import json
import gzip
import shutil
from pathlib import Path
from datetime import datetime
from typing import List, Dict, Any, Optional

class LogHandler:
    """Обработчик логов с ротацией и сжатием."""
    
    def __init__(self, log_dir: str, max_size: int = 1024 * 1024, rotate_count: int = 5):
        self.log_dir = Path(log_dir)
        self.max_size = max_size
        self.rotate_count = rotate_count
        self.log_dir.mkdir(parents=True, exist_ok=True)
    
    def _get_log_file(self) -> Path:
        """Получение текущего файла лога."""
        today = datetime.now().strftime("%Y-%m-%d")
        return self.log_dir / f"app_{today}.log"
    
    def _rotate_if_needed(self) -> None:
        """Ротация лог-файла при превышении размера."""
        log_file = self._get_log_file()
        
        if log_file.exists() and log_file.stat().st_size > self.max_size:
            timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
            rotated_file = self.log_dir / f"app_{timestamp}.log"
            shutil.move(str(log_file), str(rotated_file))
            
            # Сжатие
            with open(rotated_file, 'rb') as f:
                data = f.read()
            with gzip.open(str(rotated_file) + '.gz', 'wb') as f:
                f.write(data)
            rotated_file.unlink()
            
            # Удаление старых файлов
            all_logs = sorted(self.log_dir.glob("app_*.log*"))
            if len(all_logs) > self.rotate_count:
                for old in all_logs[:-self.rotate_count]:
                    old.unlink()
    
    def write_log(self, level: str, message: str, **kwargs) -> None:
        """Запись сообщения в лог."""
        try:
            self._rotate_if_needed()
            log_file = self._get_log_file()
            
            log_entry = {
                "timestamp": datetime.now().isoformat(),
                "level": level,
                "message": message,
                **kwargs
            }
            
            with open(log_file, 'a') as f:
                f.write(json.dumps(log_entry) + '\n')
        except PermissionError:
            raise PermissionError(f"Cannot write to log file")
        except OSError as e:
            raise OSError(f"Failed to write log: {e}")
    
    def read_logs(self, date: str, limit: int = 100) -> List[Dict[str, Any]]:
        """Чтение логов за указанную дату."""
        log_file = self.log_dir / f"app_{date}.log"
        
        if not log_file.exists():
            return []
        
        if log_file.is_dir():
            raise IsADirectoryError(f"{log_file} is a directory")
        
        logs = []
        try:
            with open(log_file, 'r') as f:
                for line in f:
                    try:
                        logs.append(json.loads(line.strip()))
                        if len(logs) >= limit:
                            break
                    except json.JSONDecodeError:
                        continue
            return logs
        except PermissionError:
            raise PermissionError(f"Cannot read log file {log_file}")
        except OSError as e:
            raise OSError(f"Failed to read logs: {e}")
    
    def clear_logs(self, date: str) -> int:
        """Удаление логов за указанную дату."""
        log_file = self.log_dir / f"app_{date}.log"
        
        if not log_file.exists():
            return 0
        
        try:
            log_file.unlink()
            return 1
        except PermissionError:
            raise PermissionError(f"Cannot delete log file {log_file}")
        except OSError as e:
            raise OSError(f"Failed to clear logs: {e}")
    
    def get_log_files(self) -> List[Path]:
        """Получение списка файлов логов."""
        try:
            return list(self.log_dir.glob("app_*.log*"))
        except PermissionError:
            raise PermissionError(f"Cannot read directory {self.log_dir}")

# csv_handler.py
import csv
import os
from pathlib import Path
from typing import List, Dict, Any, Optional

class CSVHandler:
    """Обработчик CSV-файлов."""
    
    def __init__(self, data_dir: str):
        self.data_dir = Path(data_dir)
        self.data_dir.mkdir(parents=True, exist_ok=True)
    
    def read_csv(self, filename: str, delimiter: str = ',') -> List[Dict[str, Any]]:
        """Чтение CSV-файла."""
        file_path = self.data_dir / filename
        
        if not file_path.exists():
            raise FileNotFoundError(f"File {filename} not found")
        
        if file_path.is_dir():
            raise IsADirectoryError(f"{file_path} is a directory")
        
        try:
            with open(file_path, 'r') as f:
                reader = csv.DictReader(f, delimiter=delimiter)
                return list(reader)
        except PermissionError:
            raise PermissionError(f"Cannot read {file_path}")
        except csv.Error as e:
            raise ValueError(f"CSV parsing error: {e}")
    
    def write_csv(self, filename: str, data: List[Dict[str, Any]], 
                  fieldnames: Optional[List[str]] = None) -> None:
        """Запись CSV-файла."""
        if not filename:
            raise ValueError("Filename cannot be empty")
        
        if not data:
            raise ValueError("Data cannot be empty")
        
        file_path = self.data_dir / filename
        
        if file_path.exists() and file_path.is_dir():
            raise IsADirectoryError(f"{file_path} is a directory, cannot write")
        
        fieldnames = fieldnames or list(data[0].keys())
        
        try:
            with open(file_path, 'w', newline='') as f:
                writer = csv.DictWriter(f, fieldnames=fieldnames)
                writer.writeheader()
                writer.writerows(data)
        except PermissionError:
            raise PermissionError(f"Cannot write to {file_path}")
        except OSError as e:
            raise OSError(f"Failed to write CSV: {e}")
    
    def append_csv(self, filename: str, data: List[Dict[str, Any]]) -> None:
        """Добавление данных в CSV-файл."""
        file_path = self.data_dir / filename
        
        if not file_path.exists():
            self.write_csv(filename, data)
            return
        
        if file_path.is_dir():
            raise IsADirectoryError(f"{file_path} is a directory")
        
        try:
            with open(file_path, 'r') as f:
                reader = csv.DictReader(f)
                fieldnames = reader.fieldnames
        except PermissionError:
            raise PermissionError(f"Cannot read {file_path}")
        
        try:
            with open(file_path, 'a', newline='') as f:
                writer = csv.DictWriter(f, fieldnames=fieldnames)
                writer.writerows(data)
        except PermissionError:
            raise PermissionError(f"Cannot append to {file_path}")
        except OSError as e:
            raise OSError(f"Failed to append CSV: {e}")
    
    def delete_csv(self, filename: str) -> bool:
        """Удаление CSV-файла."""
        file_path = self.data_dir / filename
        
        if not file_path.exists():
            raise FileNotFoundError(f"File {filename} not found")
        
        if file_path.is_dir():
            raise IsADirectoryError(f"{file_path} is a directory")
        
        try:
            file_path.unlink()
            return True
        except PermissionError:
            raise PermissionError(f"Cannot delete {file_path}")
        except OSError as e:
            raise OSError(f"Failed to delete file: {e}")

# json_handler.py
import json
import os
from pathlib import Path
from typing import Dict, Any, List, Optional

class JSONHandler:
    """Обработчик JSON-файлов."""
    
    def __init__(self, data_dir: str, backup_dir: Optional[str] = None):
        self.data_dir = Path(data_dir)
        self.backup_dir = Path(backup_dir) if backup_dir else self.data_dir / "backups"
        self.data_dir.mkdir(parents=True, exist_ok=True)
        self.backup_dir.mkdir(parents=True, exist_ok=True)
    
    def read_json(self, filename: str) -> Any:
        """Чтение JSON-файла."""
        file_path = self.data_dir / filename
        
        if not file_path.exists():
            raise FileNotFoundError(f"File {filename} not found")
        
        if file_path.is_dir():
            raise IsADirectoryError(f"{file_path} is a directory")
        
        try:
            with open(file_path, 'r') as f:
                return json.load(f)
        except PermissionError:
            raise PermissionError(f"Cannot read {file_path}")
        except json.JSONDecodeError as e:
            raise ValueError(f"Invalid JSON in {filename}: {e}")
    
    def write_json(self, filename: str, data: Any, indent: int = 2) -> None:
        """Запись JSON-файла."""
        if not filename:
            raise ValueError("Filename cannot be empty")
        
        file_path = self.data_dir / filename
        
        if file_path.exists() and file_path.is_dir():
            raise IsADirectoryError(f"{file_path} is a directory")
        
        try:
            with open(file_path, 'w') as f:
                json.dump(data, f, indent=indent)
        except PermissionError:
            raise PermissionError(f"Cannot write to {file_path}")
        except TypeError as e:
            raise ValueError(f"Data is not JSON serializable: {e}")
        except OSError as e:
            raise OSError(f"Failed to write JSON: {e}")
    
    def update_json(self, filename: str, updates: Dict[str, Any]) -> Dict[str, Any]:
        """Обновление данных в JSON-файле."""
        data = self.read_json(filename)
        if not isinstance(data, dict):
            raise ValueError(f"Data in {filename} is not a dictionary")
        
        data.update(updates)
        self.write_json(filename, data)
        
        # Создание бэкапа
        backup_file = self.backup_dir / f"{filename}.backup"
        try:
            with open(backup_file, 'w') as f:
                json.dump(data, f)
        except PermissionError:
            # Бэкап не критичен
            pass
        
        return data
    
    def delete_json(self, filename: str) -> bool:
        """Удаление JSON-файла."""
        file_path = self.data_dir / filename
        
        if not file_path.exists():
            raise FileNotFoundError(f"File {filename} not found")
        
        if file_path.is_dir():
            raise IsADirectoryError(f"{file_path} is a directory")
        
        try:
            file_path.unlink()
            return True
        except PermissionError:
            raise PermissionError(f"Cannot delete {file_path}")
    
    def list_json_files(self) -> List[str]:
        """Получение списка JSON-файлов."""
        try:
            return [f.name for f in self.data_dir.glob("*.json") if f.is_file()]
        except PermissionError:
            raise PermissionError(f"Cannot read directory {self.data_dir}")
    
    def merge_json(self, filename1: str, filename2: str, output_filename: str) -> None:
        """Объединение двух JSON-файлов."""
        data1 = self.read_json(filename1)
        data2 = self.read_json(filename2)
        
        if isinstance(data1, dict) and isinstance(data2, dict):
            merged = {**data1, **data2}
        elif isinstance(data1, list) and isinstance(data2, list):
            merged = data1 + data2
        else:
            raise ValueError("Files must be both dicts or both lists")
        
        self.write_json(output_filename, merged)

# image_handler.py
import os
import shutil
from pathlib import Path
from PIL import Image
from typing import List, Tuple, Optional

class ImageHandler:
    """Обработчик изображений."""
    
    def __init__(self, image_dir: str):
        self.image_dir = Path(image_dir)
        self.image_dir.mkdir(parents=True, exist_ok=True)
    
    def save_image(self, filename: str, image_data: bytes, format: str = 'PNG') -> Path:
        """Сохранение изображения."""
        if not filename:
            raise ValueError("Filename cannot be empty")
        
        file_path = self.image_dir / filename
        
        try:
            with open(file_path, 'wb') as f:
                f.write(image_data)
            return file_path
        except PermissionError:
            raise PermissionError(f"Cannot write to {file_path}")
        except OSError as e:
            raise OSError(f"Failed to save image: {e}")
    
    def load_image(self, filename: str) -> bytes:
        """Загрузка изображения."""
        file_path = self.image_dir / filename
        
        if not file_path.exists():
            raise FileNotFoundError(f"Image {filename} not found")
        
        if file_path.is_dir():
            raise IsADirectoryError(f"{file_path} is a directory")
        
        try:
            with open(file_path, 'rb') as f:
                return f.read()
        except PermissionError:
            raise PermissionError(f"Cannot read {file_path}")
        except OSError as e:
            raise OSError(f"Failed to load image: {e}")
    
    def resize_image(self, filename: str, size: Tuple[int, int], output_filename: Optional[str] = None) -> Path:
        """Изменение размера изображения."""
        file_path = self.image_dir / filename
        
        if not file_path.exists():
            raise FileNotFoundError(f"Image {filename} not found")
        
        try:
            with Image.open(file_path) as img:
                resized = img.resize(size)
                output_path = self.image_dir / (output_filename or f"resized_{filename}")
                resized.save(output_path)
                return output_path
        except PermissionError:
            raise PermissionError(f"Cannot read {file_path}")
        except OSError as e:
            raise OSError(f"Failed to resize image: {e}")
        except Exception as e:
            raise ValueError(f"Invalid image format: {e}")
    
    def delete_image(self, filename: str) -> bool:
        """Удаление изображения."""
        file_path = self.image_dir / filename
        
        if not file_path.exists():
            raise FileNotFoundError(f"Image {filename} not found")
        
        if file_path.is_dir():
            raise IsADirectoryError(f"{file_path} is a directory")
        
        try:
            file_path.unlink()
            return True
        except PermissionError:
            raise PermissionError(f"Cannot delete {file_path}")
    
    def list_images(self) -> List[str]:
        """Получение списка изображений."""
        try:
            return [f.name for f in self.image_dir.glob("*") 
                    if f.is_file() and f.suffix.lower() in ['.png', '.jpg', '.jpeg', '.gif']]
        except PermissionError:
            raise PermissionError(f"Cannot read directory {self.image_dir}")
    
    def copy_image(self, source: str, destination: str) -> Path:
        """Копирование изображения."""
        source_path = self.image_dir / source
        dest_path = self.image_dir / destination
        
        if not source_path.exists():
            raise FileNotFoundError(f"Image {source} not found")
        
        try:
            shutil.copy2(source_path, dest_path)
            return dest_path
        except PermissionError:
            raise PermissionError(f"Cannot copy {source_path} to {dest_path}")
        except OSError as e:
            raise OSError(f"Failed to copy image: {e}")

import os
import shutil
import hashlib
from pathlib import Path
from datetime import datetime
from typing import List, Dict, Any, Optional

class BackupHandler:
    """Обработчик резервных копий."""
    
    def __init__(self, source_dir: str, backup_dir: str):
        self.source_dir = Path(source_dir)
        self.backup_dir = Path(backup_dir)
        self.source_dir.mkdir(parents=True, exist_ok=True)
        self.backup_dir.mkdir(parents=True, exist_ok=True)
    
    def _calculate_hash(self, file_path: Path) -> str:
        """Вычисление MD5-хеша файла."""
        try:
            with open(file_path, 'rb') as f:
                return hashlib.md5(f.read()).hexdigest()
        except PermissionError:
            raise PermissionError(f"Cannot read {file_path}")
        except OSError as e:
            raise OSError(f"Failed to calculate hash: {e}")
    
    def create_backup(self, filename: str) -> Path:
        """Создание резервной копии файла."""
        source_path = self.source_dir / filename
        
        if not source_path.exists():
            raise FileNotFoundError(f"Source file {filename} not found")
        
        if source_path.is_dir():
            raise IsADirectoryError(f"{source_path} is a directory")
        
        timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
        backup_filename = f"{filename}.{timestamp}.backup"
        backup_path = self.backup_dir / backup_filename
        
        try:
            shutil.copy2(source_path, backup_path)
            return backup_path
        except PermissionError:
            raise PermissionError(f"Cannot create backup of {source_path}")
        except OSError as e:
            raise OSError(f"Failed to create backup: {e}")
    
    def restore_backup(self, backup_filename: str, target_filename: Optional[str] = None) -> Path:
        """Восстановление из резервной копии."""
        backup_path = self.backup_dir / backup_filename
        
        if not backup_path.exists():
            raise FileNotFoundError(f"Backup {backup_filename} not found")
        
        if backup_path.is_dir():
            raise IsADirectoryError(f"{backup_path} is a directory")
        
        target_path = self.source_dir / (target_filename or backup_filename.replace('.backup', ''))
        
        try:
            shutil.copy2(backup_path, target_path)
            return target_path
        except PermissionError:
            raise PermissionError(f"Cannot restore to {target_path}")
        except OSError as e:
            raise OSError(f"Failed to restore backup: {e}")
    
    def list_backups(self, filename: Optional[str] = None) -> List[Dict[str, Any]]:
        """Получение списка резервных копий."""
        backups = []
        pattern = f"{filename}.*.backup" if filename else "*.backup"
        
        try:
            for backup_path in self.backup_dir.glob(pattern):
                if backup_path.is_file():
                    backups.append({
                        "filename": backup_path.name,
                        "size": backup_path.stat().st_size,
                        "created": datetime.fromtimestamp(backup_path.stat().st_mtime).isoformat(),
                        "path": str(backup_path)
                    })
            return sorted(backups, key=lambda x: x["created"], reverse=True)
        except PermissionError:
            raise PermissionError(f"Cannot read directory {self.backup_dir}")
        except OSError as e:
            raise OSError(f"Failed to list backups: {e}")
    
    def delete_backup(self, backup_filename: str) -> bool:
        """Удаление резервной копии."""
        backup_path = self.backup_dir / backup_filename
        
        if not backup_path.exists():
            raise FileNotFoundError(f"Backup {backup_filename} not found")
        
        if backup_path.is_dir():
            raise IsADirectoryError(f"{backup_path} is a directory")
        
        try:
            backup_path.unlink()
            return True
        except PermissionError:
            raise PermissionError(f"Cannot delete {backup_path}")
        except OSError as e:
            raise OSError(f"Failed to delete backup: {e}")
    
    def verify_backup(self, filename: str, backup_filename: str) -> bool:
        """Проверка целостности резервной копии."""
        source_path = self.source_dir / filename
        backup_path = self.backup_dir / backup_filename
        
        if not source_path.exists():
            raise FileNotFoundError(f"Source file {filename} not found")
        
        if not backup_path.exists():
            raise FileNotFoundError(f"Backup {backup_filename} not found")
        
        try:
            source_hash = self._calculate_hash(source_path)
            backup_hash = self._calculate_hash(backup_path)
            return source_hash == backup_hash
        except (PermissionError, OSError) as e:
            raise OSError(f"Failed to verify backup: {e}")

        except PermissionError:

        except OSError as e:
            raise OSError(f"Failed to read temp file: {e}")

        except PermissionError:

        except OSError as e:
            raise OSError(f"Failed to delete temp file: {e}")

            return {s: dict(config[s]) for s in config.sections()}
        except PermissionError:
            raise PermissionError(f"Cannot read {file_path}")
        except configparser.ParsingError as e:
            raise ValueError(f"Invalid INI format: {e}")
        except OSError as e:
            raise OSError(f"Failed to load INI: {e}")
    
    def save_ini(self, filename: str, data: Dict[str, Dict[str, str]]) -> None:
        """Сохранение INI-файла."""
        if not filename:
            raise ValueError("Filename cannot be empty")
        
        file_path = self.config_dir / filename
        
        if file_path.exists() and file_path.is_dir():
            raise IsADirectoryError(f"{file_path} is a directory")
        
        try:
            config = configparser.ConfigParser()
            for section, values in data.items():
                config[section] = values
            with open(file_path, 'w') as f:
                config.write(f)
        except PermissionError:
            raise PermissionError(f"Cannot write to {file_path}")
        except OSError as e:
            raise OSError(f"Failed to save INI: {e}")
    
    def update_ini_section(self, filename: str, section: str, updates: Dict[str, str]) -> None:
        """Обновление секции в INI-файле."""
        data = self.load_ini(filename)
        if section not in data:
            data[section] = {}
        data[section].update(updates)
        self.save_ini(filename, data)
    
    def delete_ini_section(self, filename: str, section: str) -> None:
        """Удаление секции из INI-файла."""
        data = self.load_ini(filename)
        if section in data:
            del data[section]
            self.save_ini(filename, data)
    
    def delete_ini_file(self, filename: str) -> bool:
        """Удаление INI-файла."""
        file_path = self.config_dir / filename
        
        if not file_path.exists():
            raise FileNotFoundError(f"File {filename} not found")
        
        if file_path.is_dir():
            raise IsADirectoryError(f"{file_path} is a directory")
        
        try:
            file_path.unlink()
            return True
        except PermissionError:
            raise PermissionError(f"Cannot delete {file_path}")
    
    def list_ini_files(self) -> list:
        """Получение списка INI-файлов."""
        try:
            return [f.name for f in self.config_dir.glob("*.ini") if f.is_file()]
        except PermissionError:
            raise PermissionError(f"Cannot read directory {self.config_dir}")

# manifest_handler.py
import os
import json
import hashlib
from pathlib import Path
from datetime import datetime
from typing import List, Dict, Any, Optional

class ManifestHandler:
    """Обработчик файлов с манифестом для контроля целостности."""
    
    def __init__(self, data_dir: str, manifest_file: str = "manifest.json"):
        self.data_dir = Path(data_dir)
        self.manifest_file = self.data_dir / manifest_file
        self.data_dir.mkdir(parents=True, exist_ok=True)
    
    def _calculate_file_hash(self, file_path: Path) -> str:
        """Вычисление хеша файла."""
        try:
            with open(file_path, 'rb') as f:
                return hashlib.sha256(f.read()).hexdigest()
        except PermissionError:
            raise PermissionError(f"Cannot read {file_path}")
        except OSError as e:
            raise OSError(f"Failed to calculate hash: {e}")
    
    def _load_manifest(self) -> Dict[str, Dict[str, Any]]:
        """Загрузка манифеста."""
        if not self.manifest_file.exists():
            return {}
        
        try:
            with open(self.manifest_file, 'r') as f:
                return json.load(f)
        except PermissionError:
            raise PermissionError(f"Cannot read manifest {self.manifest_file}")
        except json.JSONDecodeError as e:
            raise ValueError(f"Invalid manifest: {e}")
    
    def _save_manifest(self, manifest: Dict[str, Dict[str, Any]]) -> None:
        """Сохранение манифеста."""
        try:
            with open(self.manifest_file, 'w') as f:
                json.dump(manifest, f, indent=2)
        except PermissionError:
            raise PermissionError(f"Cannot write manifest {self.manifest_file}")
        except OSError as e:
            raise OSError(f"Failed to save manifest: {e}")
    
    def add_file(self, filename: str, content: bytes) -> None:
        """Добавление файла с записью в манифест."""
        file_path = self.data_dir / filename
        
        if file_path.exists() and file_path.is_dir():
            raise IsADirectoryError(f"{file_path} is a directory")
        
        try:
            with open(file_path, 'wb') as f:
                f.write(content)
            
            file_hash = self._calculate_file_hash(file_path)
            manifest = self._load_manifest()
            manifest[filename] = {
                "hash": file_hash,
                "size": len(content),
                "added": datetime.now().isoformat()
            }
            self._save_manifest(manifest)
        except PermissionError:
            raise PermissionError(f"Cannot write to {file_path}")
        except OSError as e:
            raise OSError(f"Failed to add file: {e}")
    
    def get_file(self, filename: str) -> bytes:
        """Получение файла с проверкой по манифесту."""
        file_path = self.data_dir / filename
        
        if not file_path.exists():
            raise FileNotFoundError(f"File {filename} not found")
        
        if file_path.is_dir():
            raise IsADirectoryError(f"{file_path} is a directory")
        
        manifest = self._load_manifest()
        if filename not in manifest:
            raise ValueError(f"File {filename} not in manifest")
        
        try:
            current_hash = self._calculate_file_hash(file_path)
            if current_hash != manifest[filename]["hash"]:
                raise ValueError(f"File {filename} hash mismatch")
            
            with open(file_path, 'rb') as f:
                return f.read()
        except PermissionError:
            raise PermissionError(f"Cannot read {file_path}")
        except OSError as e:
            raise OSError(f"Failed to get file: {e}")
    
    def delete_file(self, filename: str) -> bool:
        """Удаление файла и записи из манифеста."""
        file_path = self.data_dir / filename
        
        if not file_path.exists():
            return False
        
        if file_path.is_dir():
            raise IsADirectoryError(f"{file_path} is a directory")
        
        try:
            file_path.unlink()
            manifest = self._load_manifest()
            if filename in manifest:
                del manifest[filename]
                self._save_manifest(manifest)
            return True
        except PermissionError:
            raise PermissionError(f"Cannot delete {file_path}")
        except OSError as e:
            raise OSError(f"Failed to delete file: {e}")
    
    def verify_all(self) -> Dict[str, Any]:
        """Проверка всех файлов по манифесту."""
        manifest = self._load_manifest()
        result = {"valid": [], "invalid": [], "missing": []}
        
        for filename, entry in manifest.items():
            file_path = self.data_dir / filename
            if not file_path.exists():
                result["missing"].append(filename)
                continue
            
            if file_path.is_dir():
                result["invalid"].append({"file": filename, "reason": "is_directory"})
                continue
            
            try:
                current_hash = self._calculate_file_hash(file_path)
                if current_hash == entry["hash"]:
                    result["valid"].append(filename)
                else:
                    result["invalid"].append({"file": filename, "reason": "hash_mismatch"})
            except (PermissionError, OSError) as e:
                result["invalid"].append({"file": filename, "reason": str(e)})
        
        return result
