class UserService:
    def __init__(self):
        self.users = {}
        self.next_id = 1
    
    def create_user(self, name, email, age):
        if not name or len(name) < 2:
            raise UserValidationError("Name must be at least 2 characters", "name", name)
        if not email or "@" not in email:
            raise UserValidationError("Invalid email format", "email", email)
        if age < 18 or age > 120:
            raise UserValidationError("Age must be between 18 and 120", "age", age)
        if any(u["email"] == email for u in self.users.values()):
            raise UserAlreadyExistsError(email)
        
        user_id = self.next_id
        self.users[user_id] = {"id": user_id, "name": name, "email": email, "age": age, "active": True}
        self.next_id += 1
        return user_id
    
    def get_user(self, user_id):
        if user_id not in self.users:
            raise UserNotFoundError(user_id)
        return self.users[user_id]
    
    def activate_user(self, user_id):
        user = self.get_user(user_id)
        user["active"] = True
    
    def deactivate_user(self, user_id):
        user = self.get_user(user_id)
        if not user["active"]:
            raise UserInactiveError(user_id)
        user["active"] = False
    
    def delete_user(self, user_id):
        if user_id not in self.users:
            raise UserNotFoundError(user_id)

class OrderService:
    def __init__(self):
        self.orders = {}
        self.next_id = 1
    
    def create_order(self, user_id, items):
        if not user_id:
            raise OrderValidationError("User ID is required", "user_id", user_id)
        if not items or len(items) == 0:
            raise OrderValidationError("Order must have at least one item", "items", items)
        for item in items:
            if item.get("quantity", 0) <= 0:
                raise OrderValidationError("Quantity must be positive", "quantity", item.get("quantity"))
            if item.get("price", 0) <= 0:
                raise OrderValidationError("Price must be positive", "price", item.get("price"))
        
        order_id = self.next_id
        self.orders[order_id] = {
            "id": order_id,
            "user_id": user_id,
            "items": items,
            "status": "PENDING",
            "total": sum(item["price"] * item["quantity"] for item in items)
        }
        self.next_id += 1
        return order_id
    
    def get_order(self, order_id):
        if order_id not in self.orders:
            raise OrderNotFoundError(order_id)
        return self.orders[order_id]
    
    def update_status(self, order_id, new_status):
        order = self.get_order(order_id)
        current_status = order["status"]
        
        allowed = {
            "PENDING": ["CONFIRMED", "CANCELLED"],
            "CONFIRMED": ["SHIPPED", "CANCELLED"],
            "SHIPPED": ["DELIVERED"],
            "DELIVERED": [],
            "CANCELLED": []
        }
        
        if new_status not in allowed.get(current_status, []):
            raise OrderStatusTransitionError(order_id, current_status, new_status)
        
        order["status"] = new_status
    
    def cancel_order(self, order_id):
        order = self.get_order(order_id)
        if order["status"] in ["SHIPPED", "DELIVERED"]:
            raise OrderStatusTransitionError(order_id, order["status"], "CANCELLED")
        order["status"] = "CANCELLED"

class PaymentService:
    def __init__(self):
        self.payments = {}
        self.next_id = 1
    
    def process_payment(self, order_id, amount, method):
        if not order_id:
            raise PaymentValidationError("Order ID is required", "order_id", order_id)
        if amount <= 0:
            raise PaymentValidationError("Amount must be positive", "amount", amount)
        if method not in ["card", "paypal", "wallet"]:
            raise PaymentValidationError("Invalid payment method", "method", method)
        
        payment_id = self.next_id
        self.payments[payment_id] = {
            "id": payment_id,
            "order_id": order_id,
            "amount": amount,
            "method": method,
            "status": "SUCCESS",
            "processed_at": "2026-01-01T00:00:00"
        }
        self.next_id += 1
        return payment_id
    
    def get_payment(self, payment_id):
        if payment_id not in self.payments:
            raise PaymentNotFoundError(payment_id)
        return self.payments[payment_id]
    
    def refund_payment(self, payment_id):
        payment = self.get_payment(payment_id)
        if payment["status"] != "SUCCESS":
            raise PaymentRefundError(payment_id, "Payment not successful")
        payment["status"] = "REFUNDED"
        return True
    
    def fail_payment(self, payment_id, reason):
        payment = self.get_payment(payment_id)
        if payment["status"] != "PENDING":
            raise PaymentProcessingError(payment_id, "Payment already processed")
        payment["status"] = "FAILED"
        raise PaymentProcessingError(payment_id, reason)

class ProjectService:
    def __init__(self):
        self.projects = {}
        self.next_id = 1
    
    def create_project(self, name, owner, deadline, budget):
        if not name or len(name) < 3:
            raise ProjectValidationError("Name must be at least 3 characters", "name", name)
        if not owner:
            raise ProjectValidationError("Owner is required", "owner", owner)
        if deadline < "2026-01-01":
            raise ProjectValidationError("Deadline must be in the future", "deadline", deadline)
        if budget <= 0:
            raise ProjectValidationError("Budget must be positive", "budget", budget)
        
        project_id = self.next_id
        self.projects[project_id] = {
            "id": project_id,
            "name": name,
            "owner": owner,
            "deadline": deadline,
            "budget": budget,
            "status": "PLANNING",
            "members": [owner]
        }
        self.next_id += 1
        return project_id
    
    def get_project(self, project_id):
        if project_id not in self.projects:
            raise ProjectNotFoundError(project_id)
        return self.projects[project_id]
    
    def add_member(self, project_id, user_id):
        project = self.get_project(project_id)
        if user_id in project["members"]:
            raise ProjectMemberError(project_id, user_id, "User already a member")
        project["members"].append(user_id)
    
    def remove_member(self, project_id, user_id):
        project = self.get_project(project_id)
        if user_id == project["owner"]:
            raise ProjectMemberError(project_id, user_id, "Cannot remove project owner")
        if user_id not in project["members"]:
            raise ProjectMemberError(project_id, user_id, "User not a member")
        project["members"].remove(user_id)

    def login(self, username, password):
        if username not in self.users:
            raise AuthNotFoundError(f"User {username}")
        if not self.users[username]["active"]:
            raise AuthPermissionError(username, "login")
        token = f"token_{username}_{hash(password)}"
        self.tokens[token] = username
        return token
    
    def check_permission(self, token, action):
        if token not in self.tokens:
            raise AuthTokenError(token, "invalid token")
        username = self.tokens[token]
        user = self.users.get(username)
        if not user:
            raise AuthNotFoundError(f"User {username}")
        if not user["active"]:
            raise AuthPermissionError(username, action)
        
        permissions = {
            "ADMIN": ["create", "read", "update", "delete"],
            "USER": ["read", "update"],
            "GUEST": ["read"]
        }
        if action not in permissions.get(user["role"], []):
            raise AuthPermissionError(username, action)
        return True
    
    def logout(self, token):
        if token not in self.tokens:
            raise AuthTokenError(token, "invalid token")
        del self.tokens[token]

        self.courses[course_id] = {

            "description": description,

        self.enrollments[course_id].remove(user_id)

        self.next_id = 1
        self.allowed_types = [".txt", ".pdf", ".docx", ".jpg", ".png"]
        self.max_size = 10 * 1024 * 1024  # 10 MB

        if not content:
            raise FileValidationError("Content cannot be empty", "content", None)
