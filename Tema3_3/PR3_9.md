## 2. Теоретическая справка

### 2.1. Зачем нужны моки

Моки позволяют:

| Проблема | Решение с моками |
|----------|------------------|
| Медленные тесты (запросы к API) | Подмена HTTP-запросов быстрыми моками |
| Нестабильные тесты (зависимость от сети) | Контролируемые ответы |
| Побочные эффекты (запись в БД, отправка email) | Изоляция от внешних систем |
| Зависимость от времени (тесты, которые проходят только в определённый день) | Заморозка времени |
| Тестирование ошибок (500, тайм-аут) | Симуляция ошибок |

### 2.2. Основные инструменты

| Инструмент | Назначение |
|------------|------------|
| `unittest.mock.patch` | Замена объектов и функций |
| `pytest-mock` | Удобная обёртка над `unittest.mock` |
| `responses` | Мокирование HTTP-запросов (requests) |
| `pytest_httpx` | Мокирование HTTP-запросов (httpx) |
| `monkeypatch` | Безопасная замена атрибутов и переменных окружения |
| `freezegun` | Заморозка времени |
| `pyfakefs` | Виртуальная файловая система |

# cache_service.py
import time
import json
import hashlib
from pathlib import Path
from typing import Optional, Any, Dict

class CacheService:
    """Сервис кэширования с TTL и файловым хранилищем."""
    
    def __init__(self, cache_dir: str, default_ttl: int = 300):
        self.cache_dir = Path(cache_dir)
        self.cache_dir.mkdir(parents=True, exist_ok=True)
        self.default_ttl = default_ttl
    
    def _get_cache_path(self, key: str) -> Path:
        """Получение пути к файлу кэша по ключу."""
        key_hash = hashlib.md5(key.encode()).hexdigest()
        return self.cache_dir / f"{key_hash}.cache"
    
    def get(self, key: str) -> Optional[Any]:
        """Получение значения из кэша."""
        cache_path = self._get_cache_path(key)
        
        if not cache_path.exists():
            return None
        
        try:
            with open(cache_path, 'r') as f:
                data = json.load(f)
            
            # Проверка TTL
            cached_at = data.get("_cached_at", 0)
            ttl = data.get("_ttl", self.default_ttl)
            
            if time.time() - cached_at > ttl:
                # Кэш устарел
                cache_path.unlink(missing_ok=True)
                return None
            
            return data.get("value")
        except (json.JSONDecodeError, OSError):
            return None
    
    def set(self, key: str, value: Any, ttl: Optional[int] = None) -> bool:
        """Сохранение значения в кэш."""
        cache_path = self._get_cache_path(key)
        data = {
            "value": value,
            "_cached_at": time.time(),
            "_ttl": ttl or self.default_ttl
        }
        
        try:
            with open(cache_path, 'w') as f:
                json.dump(data, f)
            return True
        except OSError:
            return False
    
    def delete(self, key: str) -> bool:
        """Удаление значения из кэша."""
        cache_path = self._get_cache_path(key)
        if cache_path.exists():
            cache_path.unlink()
            return True
        return False
    
    def clear(self) -> int:
        """Очистка всего кэша."""
        removed = 0
        for cache_file in self.cache_dir.glob("*.cache"):
            cache_file.unlink()
            removed += 1
        return removed
    
    def get_or_set(self, key: str, factory, ttl: Optional[int] = None) -> Any:
        """Получение из кэша или вычисление через фабрику."""
        value = self.get(key)
        if value is not None:
            return value
        
        value = factory()
        self.set(key, value, ttl)
        return value
    
    def get_stats(self) -> Dict[str, Any]:
        """Получение статистики кэша."""
        files = list(self.cache_dir.glob("*.cache"))
        total_size = sum(f.stat().st_size for f in files)
        
        return {
            "total_entries": len(files),
            "total_size_bytes": total_size,
            "cache_dir": str(self.cache_dir)
        }

# auth_service.py
import requests
import time
import jwt
from typing import Optional, Dict, Any

class AuthService:
    """Сервис авторизации с OAuth2."""
    
    def __init__(self, auth_url: str, client_id: str, client_secret: str):
        self.auth_url = auth_url
        self.client_id = client_id
        self.client_secret = client_secret
        self.access_token = None
        self.refresh_token = None
        self.token_expires_at = 0
    
    def get_access_token(self) -> str:
        """Получение access token."""
        if self.access_token and time.time() < self.token_expires_at:
            return self.access_token
        
        # Запрос нового токена
        data = {
            "grant_type": "client_credentials",
            "client_id": self.client_id,
            "client_secret": self.client_secret
        }
        
        try:
            response = requests.post(self.auth_url, data=data, timeout=5)
            response.raise_for_status()
            token_data = response.json()
            
            self.access_token = token_data["access_token"]
            self.refresh_token = token_data.get("refresh_token")
            self.token_expires_at = time.time() + token_data.get("expires_in", 3600)
            
            return self.access_token
        except requests.exceptions.Timeout:
            raise Exception("Auth service timeout")
        except requests.exceptions.HTTPError as e:
            if e.response.status_code == 400:
                raise Exception("Invalid client credentials")
            raise Exception(f"Auth service error: {e}")
        except requests.exceptions.RequestException as e:
            raise Exception(f"Auth request failed: {e}")
    
    def refresh_access_token(self) -> str:
        """Обновление access token по refresh token."""
        if not self.refresh_token:
            return self.get_access_token()
        
        data = {
            "grant_type": "refresh_token",
            "refresh_token": self.refresh_token,
            "client_id": self.client_id,
            "client_secret": self.client_secret
        }
        
        try:
            response = requests.post(self.auth_url, data=data, timeout=5)
            response.raise_for_status()
            token_data = response.json()
            
            self.access_token = token_data["access_token"]
            self.refresh_token = token_data.get("refresh_token", self.refresh_token)
            self.token_expires_at = time.time() + token_data.get("expires_in", 3600)
            
            return self.access_token
        except requests.exceptions.Timeout:
            raise Exception("Auth service timeout")
        except requests.exceptions.HTTPError as e:
            if e.response.status_code == 400:
                raise Exception("Invalid refresh token")
            raise Exception(f"Auth service error: {e}")
        except requests.exceptions.RequestException as e:
            raise Exception(f"Auth request failed: {e}")
    
    def validate_token(self, token: str) -> bool:
        """Проверка валидности токена."""
        try:
            # Имитация проверки токена
            response = requests.post(
                f"{self.auth_url}/introspect",
                data={"token": token},
                timeout=5
            )
            response.raise_for_status()
            return response.json().get("active", False)
        except requests.exceptions.RequestException:
            return False
    
    def get_user_info(self, token: str) -> Optional[Dict[str, Any]]:
        """Получение информации о пользователе по токену."""
        try:
            response = requests.get(
                f"{self.auth_url}/userinfo",
                headers={"Authorization": f"Bearer {token}"},
                timeout=5
            )
            response.raise_for_status()
            return response.json()
        except requests.exceptions.HTTPError as e:
            if e.response.status_code == 401:
                return None
            raise
        except requests.exceptions.RequestException:
            return None

        self.rotate_count = rotate_count
        self.current_log_file = None
    
    def _get_log_file(self) -> Path:
        """Получение текущего файла лога."""
        today = datetime.now().strftime("%Y-%m-%d")
        return self.log_dir / f"app_{today}.log"
    
    def _rotate_if_needed(self) -> None:
        """Ротация файла лога при превышении размера."""
        log_file = self._get_log_file()
        
        if log_file.exists() and log_file.stat().st_size > self.max_file_size:
            # Создание ротированного файла
            timestamp = int(time.time())
            rotated_file = self.log_dir / f"app_{timestamp}.log"
            log_file.rename(rotated_file)
            
            # Сжатие старого файла
            with open(rotated_file, 'rb') as f:
                data = f.read()
            with gzip.open(str(rotated_file) + '.gz', 'wb') as f:
                f.write(data)
            rotated_file.unlink()
            
            # Удаление старых файлов (оставляем только rotate_count)
            all_logs = sorted(self.log_dir.glob("app_*.log*"))
            if len(all_logs) > self.rotate_count:
                for old in all_logs[:-self.rotate_count]:
                    old.unlink()
    
    def log(self, level: str, message: str, **kwargs) -> None:
        """Запись сообщения в лог."""
        self._rotate_if_needed()
        
        log_file = self._get_log_file()
        log_entry = {
            "timestamp": datetime.now().isoformat(),
            "level": level,
            "message": message,
            **kwargs
        }
        
        with open(log_file, 'a') as f:
            f.write(json.dumps(log_entry) + '\n')
    
    def log_error(self, message: str, exception: Optional[Exception] = None, **kwargs) -> None:
        """Запись ошибки в лог."""
        error_data = {
            "message": message,
            "exception": str(exception) if exception else None,
            **kwargs
        }
        self.log("ERROR", message, **error_data)
    
    def get_logs(self, date: str, limit: int = 100) -> List[Dict[str, Any]]:
        """Получение логов за определённый день."""
        log_file = self.log_dir / f"app_{date}.log"
        
        if not log_file.exists():
            return []
        
        logs = []
        with open(log_file, 'r') as f:
            for line in f:
                try:
                    logs.append(json.loads(line.strip()))
                    if len(logs) >= limit:
                        break
                except json.JSONDecodeError:
                    continue
        
        return logs
    
    def get_logs_by_level(self, level: str, limit: int = 100) -> List[Dict[str, Any]]:
        """Получение логов по уровню за сегодня."""
        log_file = self._get_log_file()
        
        if not log_file.exists():
            return []
        
        logs = []
        with open(log_file, 'r') as f:
            for line in f:
                try:
                    data = json.loads(line.strip())
                    if data.get("level") == level:
                        logs.append(data)
                        if len(logs) >= limit:
                            break
                except json.JSONDecodeError:
                    continue
        
        return logs
    
    def get_rotated_logs(self) -> List[Path]:
        """Получение списка ротированных файлов логов."""
        return sorted(self.log_dir.glob("app_*.log.gz"))

import os
import json
import hashlib
import requests
from pathlib import Path
from typing import Optional, Dict, Any, List

class FileProcessingService:
    """Сервис обработки файлов с внешним API."""
    
    def __init__(self, api_url: str, api_key: str, work_dir: str):
        self.api_url = api_url
        self.api_key = api_key
        self.work_dir = Path(work_dir)
        self.work_dir.mkdir(parents=True, exist_ok=True)
        self.session = requests.Session()
        self.session.headers.update({"Authorization": f"Bearer {api_key}"})
    
    def process_file(self, file_path: Path) -> Dict[str, Any]:
        """Обработка файла через внешний API."""
        if not file_path.exists():
            raise FileNotFoundError(f"File not found: {file_path}")
        
        # Чтение файла
        with open(file_path, 'rb') as f:
            content = f.read()
        
        # Вычисление хеша
        file_hash = hashlib.sha256(content).hexdigest()
        
        # Отправка на обработку
        url = f"{self.api_url}/process"
        files = {"file": (file_path.name, content)}
        
        try:
            response = self.session.post(url, files=files, timeout=30)
            response.raise_for_status()
            result = response.json()
            
            # Сохранение результата
            result_file = self.work_dir / f"{file_path.stem}_result.json"
            with open(result_file, 'w') as f:
                json.dump({"hash": file_hash, "result": result}, f)
            
            return result
        except requests.exceptions.Timeout:
            raise Exception("Processing API timeout")
        except requests.exceptions.HTTPError as e:
            if e.response.status_code == 413:
                raise Exception("File too large")
            elif e.response.status_code == 415:
                raise Exception("Unsupported file type")
            raise Exception(f"Processing API error: {e}")
        except requests.exceptions.RequestException as e:
            raise Exception(f"Processing request failed: {e}")
    
    def process_multiple_files(self, file_paths: List[Path]) -> Dict[str, Any]:
        """Обработка нескольких файлов."""
        results = {}
        for file_path in file_paths:
            try:
                results[file_path.name] = self.process_file(file_path)
            except Exception as e:
                results[file_path.name] = {"error": str(e)}
        return results
    
    def get_processing_status(self, job_id: str) -> Dict[str, Any]:
        """Получение статуса обработки."""
        url = f"{self.api_url}/status/{job_id}"
        
        try:
            response = self.session.get(url, timeout=5)
            response.raise_for_status()
            return response.json()
        except requests.exceptions.Timeout:
            raise Exception("Status API timeout")
        except requests.exceptions.HTTPError as e:
            if e.response.status_code == 404:
                raise Exception(f"Job {job_id} not found")
            raise Exception(f"Status API error: {e}")
        except requests.exceptions.RequestException as e:
            raise Exception(f"Status request failed: {e}")
    
    def cleanup(self, max_age_days: int = 7) -> int:
        """Очистка старых файлов."""
        import time
        deleted = 0
        cutoff = time.time() - (max_age_days * 24 * 60 * 60)
        
        for file_path in self.work_dir.glob("*"):
            if file_path.stat().st_mtime < cutoff:
                file_path.unlink()
                deleted += 1
        
        return deleted

        self.backup_dir = Path(backup_dir)
        self.storage_url = storage_url
        self.api_key = api_key

        self.backup_dir.mkdir(parents=True, exist_ok=True)

        self.session.headers.update({"Authorization": f"Bearer {api_key}"})
    
    def create_backup(self, file_name: str, compress: bool = False) -> Dict[str, Any]:
        """Создание локальной копии файла."""
        source_file = self.source_dir / file_name
        
        if not source_file.exists():
            raise FileNotFoundError(f"Source file not found: {file_name}")
        
        # Создание бэкапа
        timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
        backup_name = f"{file_name}.{timestamp}.backup"
        backup_path = self.backup_dir / backup_name
        
        if compress:
            import gzip
            with open(source_file, 'rb') as f:
                data = f.read()
            with gzip.open(backup_path, 'wb') as f:
                f.write(data)
            backup_path = backup_path.with_suffix('.gz')
        else:
            shutil.copy2(source_file, backup_path)
        
        return {
            "backup_file": backup_path.name,
            "backup_path": str(backup_path),
            "created_at": timestamp,
            "size": backup_path.stat().st_size,
            "compressed": compress
        }
    
    def restore_backup(self, backup_name: str, target_path: Optional[Path] = None) -> Path:
        """Восстановление из бэкапа."""
        backup_path = self.backup_dir / backup_name
        
        if not backup_path.exists():
            raise FileNotFoundError(f"Backup file not found: {backup_name}")
        
        target = target_path or self.source_dir / backup_path.stem.replace('.backup', '')
        
        if backup_path.suffix == '.gz':
            import gzip
            with gzip.open(backup_path, 'rb') as f:
                data = f.read()
            with open(target, 'wb') as f:
                f.write(data)
        else:
            shutil.copy2(backup_path, target)
        
        return target
    
    def upload_to_storage(self, backup_name: str) -> Dict[str, Any]:
        """Загрузка бэкапа в облачное хранилище."""
        backup_path = self.backup_dir / backup_name

            raise FileNotFoundError(f"Backup file not found: {backup_name}")
        
        url = f"{self.storage_url}/api/v1/upload"

        deleted = 0
        cutoff = time.time() - (max_age_days * 24 * 60 * 60)
        
        for backup_path in self.backup_dir.glob("*.backup*"):

                deleted += 1
        
        return deleted

        except requests.exceptions.Timeout:
            raise Exception("Monitoring API timeout")
        except requests.exceptions.HTTPError as e:
            if e.response.status_code == 400:
                raise Exception("Invalid metrics data")
            raise Exception(f"Monitoring API error: {e}")
        except requests.exceptions.RequestException as e:
            raise Exception(f"Monitoring request failed: {e}")
    
    def run(self, duration: Optional[int] = None) -> List[Dict[str, Any]]:
        """Запуск сбора и отправки метрик."""
        start_time = time.time()
        results = []
        
        while True:
            if duration and (time.time() - start_time) > duration:
                break
            
            try:
                metrics = self.collect_metrics()
                self.send_metrics(metrics)
                results.append(metrics)
            except Exception as e:
                # Логирование ошибки
                results.append({"error": str(e)})
            
            time.sleep(self.report_interval)
        
        return results
    
    def check_health(self) -> Dict[str, Any]:
        """Проверка здоровья системы."""
        url = f"{self.api_url}/api/v1/health"
        
        try:
            response = self.session.get(url, timeout=5)
            response.raise_for_status()
            return {"status": "healthy", "details": response.json()}
        except requests.exceptions.Timeout:
            return {"status": "unhealthy", "reason": "timeout"}
        except requests.exceptions.HTTPError as e:
            return {"status": "unhealthy", "reason": f"HTTP {e.response.status_code}"}
        except requests.exceptions.RequestException as e:
            return {"status": "unhealthy", "reason": str(e)}

# scraping_service.py
import time
import json
import requests
from bs4 import BeautifulSoup
from pathlib import Path
from typing import Dict, Any, List, Optional

class ScrapingService:
    """Сервис веб-скрапинга."""
    
    def __init__(self, cache_dir: str, user_agent: str = "Mozilla/5.0"):
        self.cache_dir = Path(cache_dir)
        self.cache_dir.mkdir(parents=True, exist_ok=True)
        self.user_agent = user_agent
        self.session = requests.Session()
        self.session.headers.update({"User-Agent": user_agent})
    
    def fetch_page(self, url: str, use_cache: bool = True) -> str:
        """Получение HTML страницы."""
        if use_cache:
            cached = self._get_cached(url)
            if cached:
                return cached
        
        try:
            response = self.session.get(url, timeout=10)
            response.raise_for_status()
            html = response.text
            
            if use_cache:
                self._cache_page(url, html)
            
            return html
        except requests.exceptions.Timeout:
            raise Exception(f"Timeout fetching {url}")
        except requests.exceptions.HTTPError as e:
            if e.response.status_code == 404:
                raise Exception(f"Page not found: {url}")
            if e.response.status_code == 403:
                raise Exception(f"Access forbidden: {url}")
            raise Exception(f"HTTP error: {e}")
        except requests.exceptions.RequestException as e:
            raise Exception(f"Request failed: {e}")
    
    def _get_cached(self, url: str) -> Optional[str]:
        """Получение страницы из кэша."""
        cache_file = self._get_cache_path(url)
        if cache_file.exists():
            with open(cache_file, 'r') as f:
                data = json.load(f)
            return data.get("html")
        return None
    
    def _cache_page(self, url: str, html: str) -> None:
        """Сохранение страницы в кэш."""
        cache_file = self._get_cache_path(url)
        with open(cache_file, 'w') as f:
            json.dump({"url": url, "html": html, "cached_at": time.time()}, f)
    
    def _get_cache_path(self, url: str) -> Path:
        """Получение пути к файлу кэша."""
        import hashlib
        url_hash = hashlib.md5(url.encode()).hexdigest()
        return self.cache_dir / f"{url_hash}.html"
    
    def scrape_title(self, url: str) -> str:
        """Извлечение заголовка страницы."""
        html = self.fetch_page(url)
        soup = BeautifulSoup(html, 'html.parser')
        title_tag = soup.find('title')
        return title_tag.text.strip() if title_tag else ""
    
    def scrape_links(self, url: str, limit: int = 10) -> List[str]:
        """Извлечение ссылок со страницы."""
        html = self.fetch_page(url)
        soup = BeautifulSoup(html, 'html.parser')
        links = []
        for a in soup.find_all('a', href=True):
            href = a['href']
            if href.startswith('http'):
                links.append(href)
                if len(links) >= limit:
                    break
        return links
    
    def scrape_meta_description(self, url: str) -> Optional[str]:
        """Извлечение meta description."""
        html = self.fetch_page(url)
        soup = BeautifulSoup(html, 'html.parser')
        meta = soup.find('meta', attrs={'name': 'description'})
        return meta.get('content') if meta else None
    
    def scrape_multiple(self, urls: List[str]) -> Dict[str, Any]:
        """Скрапинг нескольких страниц."""
        results = {}
        for url in urls:
            try:
                results[url] = {
                    "title": self.scrape_title(url),
                    "description": self.scrape_meta_description(url)
                }
            except Exception as e:
                results[url] = {"error": str(e)}
        return results

# jwt_auth_service.py
import time
import jwt
import json
import requests
from pathlib import Path
from typing import Optional, Dict, Any

class JWTAuthService:
    """Сервис аутентификации с JWT токенами."""
    
    def __init__(self, auth_url: str, client_id: str, client_secret: str, cache_dir: Optional[str] = None):
        self.auth_url = auth_url
        self.client_id = client_id
        self.client_secret = client_secret
        self.cache_dir = Path(cache_dir) if cache_dir else None
        
        if self.cache_dir:
            self.cache_dir.mkdir(parents=True, exist_ok=True)
        
        self.access_token = None
        self.refresh_token = None
        self.token_expires_at = 0
    
    def _get_token_from_cache(self) -> Optional[str]:
        """Получение токена из кэша."""
        if not self.cache_dir:
            return None
        
        cache_file = self.cache_dir / "token_cache.json"
        if not cache_file.exists():
            return None
        
        try:
            with open(cache_file, 'r') as f:
                data = json.load(f)
            
            if data.get("expires_at", 0) > time.time():
                return data.get("access_token")
        except (json.JSONDecodeError, OSError):
            return None
        
        return None
    
    def _save_token_to_cache(self, access_token: str, expires_at: float) -> None:
        """Сохранение токена в кэш."""
        if not self.cache_dir:
            return
        
        cache_file = self.cache_dir / "token_cache.json"
        try:
            with open(cache_file, 'w') as f:
                json.dump({
                    "access_token": access_token,
                    "expires_at": expires_at
                }, f)
        except OSError:
            pass
    
    def get_access_token(self) -> str:
        """Получение access token."""
        if self.access_token and time.time() < self.token_expires_at:
            return self.access_token
        
        cached_token = self._get_token_from_cache()
        if cached_token:
            self.access_token = cached_token
            return cached_token
        
        data = {
            "grant_type": "client_credentials",
            "client_id": self.client_id,
            "client_secret": self.client_secret
        }
        
        try:
            response = requests.post(self.auth_url, data=data, timeout=5)
            response.raise_for_status()
            token_data = response.json()
            
            self.access_token = token_data["access_token"]
            self.refresh_token = token_data.get("refresh_token")
            self.token_expires_at = time.time() + token_data.get("expires_in", 3600)
            
            self._save_token_to_cache(self.access_token, self.token_expires_at)
            
            return self.access_token
        except requests.exceptions.Timeout:
            raise Exception("Auth service timeout")
        except requests.exceptions.HTTPError as e:
            if e.response.status_code == 400:
                raise Exception("Invalid client credentials")
            raise Exception(f"Auth service error: {e}")
        except requests.exceptions.RequestException as e:
            raise Exception(f"Auth request failed: {e}")
    
    def introspect_token(self, token: str) -> Dict[str, Any]:
        """Проверка валидности токена."""
        url = f"{self.auth_url}/introspect"
        data = {"token": token}
        
        try:
            response = requests.post(url, data=data, timeout=5)
            response.raise_for_status()
            return response.json()
        except requests.exceptions.Timeout:
            raise Exception("Introspection timeout")
        except requests.exceptions.HTTPError as e:
            raise Exception(f"Introspection error: {e}")
        except requests.exceptions.RequestException as e:
            raise Exception(f"Introspection request failed: {e}")
    
    def decode_jwt(self, token: str) -> Dict[str, Any]:
        """Декодирование JWT (без проверки подписи)."""
        try:
            return jwt.decode(token, options={"verify_signature": False})
        except jwt.InvalidTokenError as e:
            raise Exception(f"Invalid JWT: {e}")
    
    def validate_jwt(self, token: str, secret: str) -> bool:
        """Проверка JWT с секретом."""
        try:
            jwt.decode(token, secret, algorithms=["HS256"])
            return True
        except jwt.InvalidTokenError:
            return False
