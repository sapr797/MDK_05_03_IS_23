## 2. Теоретическая справка

### 2.1. Что такое комплексное мокирование

**Комплексное мокирование** — это техника, при которой в одном тесте одновременно заменяются несколько внешних зависимостей:

| Зависимость | Инструмент |
|-------------|------------|
| HTTP-запросы | `responses`, `pytest_httpx`, `patch` |
| Время | `freezegun`, `monkeypatch` |
| Файловая система | `tmp_path`, `pyfakefs`, `monkeypatch` |
| База данных | `patch` для подключения/запросов |
| Внешние сервисы | `patch` для клиентов |
| Переменные окружения | `monkeypatch.setenv` |

### 2.2. Паттерны комплексного мокирования

1. **Комбинированный патч:** Использование нескольких `patch` декораторов.
2. **Фикстуры с моками:** Создание фикстур, которые настраивают все зависимости.
3. **Автоиспользование моков:** Использование `mocker` для автоматического управления.
4. **Верификация всех вызовов:** Проверка, что все зависимости вызваны корректно.

import json
import csv
import requests
import os
from pathlib import Path
from datetime import datetime
from typing import Dict, Any, List, Optional

class ImportService:
    """Сервис импорта данных из файлов с валидацией через внешний API."""
    
    def __init__(self, api_url: str, api_key: str, import_dir: str):
        self.api_url = api_url
        self.api_key = api_key
        self.import_dir = Path(import_dir)
        self.import_dir.mkdir(parents=True, exist_ok=True)
        self.session = requests.Session()
        self.session.headers.update({
            "Authorization": f"Bearer {api_key}",
            "Content-Type": "application/json"
        })
    
    def validate_data(self, data: List[Dict[str, Any]]) -> Dict[str, Any]:
        """Валидация данных через внешний API."""
        url = f"{self.api_url}/api/v1/validate"
        
        try:
            response = self.session.post(url, json={"data": data}, timeout=10)
            response.raise_for_status()
            return response.json()
        except requests.exceptions.Timeout:
            raise Exception("Validation API timeout")
        except requests.exceptions.HTTPError as e:
            if e.response.status_code == 400:
                raise Exception("Invalid data format")
            raise Exception(f"Validation API error: {e}")
        except requests.exceptions.RequestException as e:
            raise Exception(f"Validation request failed: {e}")
    
    def import_from_csv(self, filename: str) -> Dict[str, Any]:
        """Импорт данных из CSV файла."""
        file_path = self.import_dir / filename
        
        if not file_path.exists():
            raise FileNotFoundError(f"File not found: {filename}")
        
        # Чтение CSV
        data = []
        with open(file_path, 'r') as f:
            reader = csv.DictReader(f)
            for row in reader:
                data.append(row)
        
        if not data:
            raise ValueError("CSV file is empty")
        
        # Валидация через API
        validation_result = self.validate_data(data)
        
        # Сохранение результата импорта
        timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
        result_file = self.import_dir / f"import_{timestamp}.json"
        result_data = {
            "filename": filename,
            "timestamp": timestamp,
            "total_rows": len(data),
            "valid_rows": validation_result.get("valid_count", 0),
            "invalid_rows": validation_result.get("invalid_count", 0),
            "errors": validation_result.get("errors", [])
        }
        
        with open(result_file, 'w') as f:
            json.dump(result_data, f)
        
        return result_data
    
    def import_from_json(self, filename: str) -> Dict[str, Any]:
        """Импорт данных из JSON файла."""
        file_path = self.import_dir / filename
        
        if not file_path.exists():
            raise FileNotFoundError(f"File not found: {filename}")
        
        with open(file_path, 'r') as f:
            data = json.load(f)
        
        if not data:
            raise ValueError("JSON file is empty")
        
        if not isinstance(data, list):
            data = [data]
        
        # Валидация через API
        validation_result = self.validate_data(data)
        
        # Сохранение результата
        timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
        result_file = self.import_dir / f"import_{timestamp}.json"
        result_data = {
            "filename": filename,
            "timestamp": timestamp,
            "total_rows": len(data),
            "valid_rows": validation_result.get("valid_count", 0),
            "invalid_rows": validation_result.get("invalid_count", 0),
            "errors": validation_result.get("errors", [])
        }
        
        with open(result_file, 'w') as f:
            json.dump(result_data, f)
        
        return result_data
    
    def import_multiple(self, filenames: List[str]) -> List[Dict[str, Any]]:
        """Импорт нескольких файлов."""
        results = []
        for filename in filenames:
            try:
                if filename.endswith('.csv'):
                    results.append(self.import_from_csv(filename))
                elif filename.endswith('.json'):
                    results.append(self.import_from_json(filename))
                else:
                    results.append({"error": f"Unsupported file type: {filename}"})
            except Exception as e:
                results.append({"error": str(e), "filename": filename})
        return results
    
    def get_import_history(self, limit: int = 10) -> List[Dict[str, Any]]:
        """Получение истории импортов."""
        import_files = sorted(
            self.import_dir.glob("import_*.json"),
            key=lambda x: x.stat().st_mtime,
            reverse=True
        )[:limit]
        
        history = []
        for file_path in import_files:
            try:
                with open(file_path, 'r') as f:
                    history.append(json.load(f))
            except json.JSONDecodeError:
                continue
        
        return history

import json
import time
import requests
import smtplib
from pathlib import Path
from email.mime.text import MIMEText
from email.mime.multipart import MIMEMultipart
from datetime import datetime
from typing import Dict, Any, List, Optional

class ReportService:
    """Сервис генерации и отправки отчётов."""
    
    def __init__(self, api_url: str, api_key: str, smtp_server: str, smtp_port: int,
                 smtp_user: str, smtp_password: str, report_dir: str):
        self.api_url = api_url
        self.api_key = api_key
        self.smtp_server = smtp_server
        self.smtp_port = smtp_port
        self.smtp_user = smtp_user
        self.smtp_password = smtp_password
        self.report_dir = Path(report_dir)
        self.report_dir.mkdir(parents=True, exist_ok=True)
        self.session = requests.Session()
        self.session.headers.update({
            "Authorization": f"Bearer {api_key}",
            "Content-Type": "application/json"
        })
    
    def generate_report_data(self, report_type: str, params: Dict[str, Any]) -> Dict[str, Any]:
        """Генерация данных для отчёта через API."""
        url = f"{self.api_url}/api/v1/reports/{report_type}"
        
        try:
            response = self.session.post(url, json=params, timeout=30)
            response.raise_for_status()
            return response.json()
        except requests.exceptions.Timeout:
            raise Exception("Report API timeout")
        except requests.exceptions.HTTPError as e:
            if e.response.status_code == 404:
                raise Exception(f"Report type '{report_type}' not found")
            raise Exception(f"Report API error: {e}")
        except requests.exceptions.RequestException as e:
            raise Exception(f"Report request failed: {e}")
    
    def save_report(self, report_type: str, data: Dict[str, Any]) -> Path:
        """Сохранение отчёта в файл."""
        timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
        filename = f"report_{report_type}_{timestamp}.json"
        file_path = self.report_dir / filename
        
        with open(file_path, 'w') as f:
            json.dump({"report_type": report_type, "data": data, "generated_at": timestamp}, f)
        
        return file_path
    
    def send_email_report(self, recipient: str, report_type: str, params: Dict[str, Any]) -> bool:
        """Генерация и отправка отчёта по email."""
        try:
            # Генерация данных
            data = self.generate_report_data(report_type, params)
            
            # Сохранение отчёта
            file_path = self.save_report(report_type, data)
            
            # Отправка email
            msg = MIMEMultipart()
            msg['From'] = self.smtp_user
            msg['To'] = recipient
            msg['Subject'] = f"Report: {report_type}"
            
            body = f"Report generated at {datetime.now()}\nData: {json.dumps(data, indent=2)}"
            msg.attach(MIMEText(body, 'plain'))
            
            with open(file_path, 'r') as f:
                attachment = MIMEText(f.read())
                attachment.add_header('Content-Disposition', 'attachment', filename=file_path.name)
                msg.attach(attachment)
            
            # Отправка через SMTP
            with smtplib.SMTP(self.smtp_server, self.smtp_port) as server:
                server.starttls()
                server.login(self.smtp_user, self.smtp_password)
                server.send_message(msg)
            
            return True
        
        except smtplib.SMTPAuthenticationError:
            raise Exception("SMTP authentication failed")
        except smtplib.SMTPException as e:
            raise Exception(f"SMTP error: {e}")
        except Exception as e:
            raise Exception(f"Report generation failed: {e}")
    
    def send_multiple_reports(self, recipients: List[str], report_type: str, params: Dict[str, Any]) -> Dict[str, Any]:
        """Отправка отчёта нескольким получателям."""
        results = {"success": [], "failed": []}
        
        for recipient in recipients:
            try:
                self.send_email_report(recipient, report_type, params)
                results["success"].append(recipient)
            except Exception as e:
                results["failed"].append({"recipient": recipient, "error": str(e)})
        
        return results
    
    def get_report_files(self, limit: int = 10) -> List[Path]:
        """Получение списка файлов отчётов."""
        return sorted(
            self.report_dir.glob("report_*.json"),
            key=lambda x: x.stat().st_mtime,
            reverse=True
        )[:limit]

# sync_service.py
import json
import time
import hashlib
import requests
import sqlite3
from pathlib import Path
from datetime import datetime
from typing import Dict, Any, List, Optional

class SyncService:
    """Сервис синхронизации данных между локальной БД и удалённым API."""
    
    def __init__(self, api_url: str, api_key: str, db_path: str):
        self.api_url = api_url
        self.api_key = api_key
        self.db_path = Path(db_path)
        self.session = requests.Session()
        self.session.headers.update({
            "Authorization": f"Bearer {api_key}",
            "Content-Type": "application/json"
        })
        self._init_db()
    
    def _init_db(self) -> None:
        """Инициализация локальной базы данных."""
        self.db_path.parent.mkdir(parents=True, exist_ok=True)
        
        conn = sqlite3.connect(str(self.db_path))
        cursor = conn.cursor()
        cursor.execute('''
            CREATE TABLE IF NOT EXISTS records (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                remote_id TEXT UNIQUE,
                data TEXT,
                hash TEXT,
                synced_at TIMESTAMP
            )
        ''')
        cursor.execute('''
            CREATE TABLE IF NOT EXISTS sync_log (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                operation TEXT,
                record_id TEXT,
                status TEXT,
                error TEXT,
                timestamp TIMESTAMP
            )
        ''')
        conn.commit()
        conn.close()
    
    def fetch_remote_data(self, since: Optional[str] = None) -> List[Dict[str, Any]]:
        """Получение данных из удалённого API."""
        url = f"{self.api_url}/api/v1/data"
        params = {"since": since} if since else {}
        
        try:
            response = self.session.get(url, params=params, timeout=10)
            response.raise_for_status()
            return response.json().get("data", [])
        except requests.exceptions.Timeout:
            raise Exception("API timeout")
        except requests.exceptions.HTTPError as e:
            raise Exception(f"API error: {e}")
        except requests.exceptions.RequestException as e:
            raise Exception(f"Request failed: {e}")
    
    def push_local_data(self, data: List[Dict[str, Any]]) -> Dict[str, Any]:
        """Отправка локальных данных в удалённый API."""
        url = f"{self.api_url}/api/v1/data"
        
        try:
            response = self.session.post(url, json={"data": data}, timeout=10)
            response.raise_for_status()
            return response.json()
        except requests.exceptions.Timeout:
            raise Exception("API timeout")
        except requests.exceptions.HTTPError as e:
            raise Exception(f"API error: {e}")
        except requests.exceptions.RequestException as e:
            raise Exception(f"Request failed: {e}")
    
    def _calculate_hash(self, data: Dict[str, Any]) -> str:
        """Вычисление хеша данных."""
        return hashlib.sha256(json.dumps(data, sort_keys=True).encode()).hexdigest()
    
    def sync_pull(self) -> Dict[str, Any]:
        """Синхронизация: загрузка данных из удалённого API."""
        conn = sqlite3.connect(str(self.db_path))
        cursor = conn.cursor()
        
        try:
            # Получение последней синхронизации
            cursor.execute("SELECT MAX(synced_at) FROM records")
            last_sync = cursor.fetchone()[0]
            
            # Загрузка данных из API
            remote_data = self.fetch_remote_data(last_sync)
            
            added = 0
            updated = 0
            
            for record in remote_data:
                record_id = record.get("id")
                data_hash = self._calculate_hash(record)
                
                # Проверка существования записи
                cursor.execute("SELECT id, hash FROM records WHERE remote_id = ?", (record_id,))
                existing = cursor.fetchone()
                
                if existing:
                    if existing[1] != data_hash:
                        # Обновление
                        cursor.execute(
                            "UPDATE records SET data = ?, hash = ?, synced_at = ? WHERE remote_id = ?",
                            (json.dumps(record), data_hash, datetime.now(), record_id)
                        )
                        updated += 1
                else:
                    # Добавление
                    cursor.execute(
                        "INSERT INTO records (remote_id, data, hash, synced_at) VALUES (?, ?, ?, ?)",
                        (record_id, json.dumps(record), data_hash, datetime.now())
                    )
                    added += 1
            
            conn.commit()
            
            # Логирование
            cursor.execute(
                "INSERT INTO sync_log (operation, status, timestamp) VALUES (?, ?, ?)",
                ("PULL", "SUCCESS", datetime.now())
            )
            conn.commit()
            
            return {"added": added, "updated": updated, "total": len(remote_data)}
        
        except Exception as e:
            cursor.execute(
                "INSERT INTO sync_log (operation, status, error, timestamp) VALUES (?, ?, ?, ?)",
                ("PULL", "FAILED", str(e), datetime.now())
            )
            conn.commit()
            raise
        finally:
            conn.close()
    
    def sync_push(self) -> Dict[str, Any]:
        """Синхронизация: отправка локальных данных в удалённый API."""
        conn = sqlite3.connect(str(self.db_path))
        cursor = conn.cursor()
        
        try:
            # Получение всех локальных записей
            cursor.execute("SELECT remote_id, data FROM records")
            local_data = []
            for row in cursor.fetchall():
                local_data.append({
                    "id": row[0],
                    "data": json.loads(row[1])
                })
            
            if not local_data:
                return {"sent": 0, "message": "No data to sync"}
            
            # Отправка в API
            result = self.push_local_data(local_data)
            
            # Логирование
            cursor.execute(
                "INSERT INTO sync_log (operation, status, timestamp) VALUES (?, ?, ?)",
                ("PUSH", "SUCCESS", datetime.now())
            )
            conn.commit()
            
            return {"sent": len(local_data), "result": result}
        
        except Exception as e:
            cursor.execute(
                "INSERT INTO sync_log (operation, status, error, timestamp) VALUES (?, ?, ?)",
                ("PUSH", "FAILED", str(e), datetime.now())
            )
            conn.commit()
            raise
        finally:
            conn.close()
    
    def get_sync_log(self, limit: int = 20) -> List[Dict[str, Any]]:
        """Получение лога синхронизаций."""
        conn = sqlite3.connect(str(self.db_path))
        cursor = conn.cursor()
        
        cursor.execute(
            "SELECT operation, status, error, timestamp FROM sync_log ORDER BY id DESC LIMIT ?",
            (limit,)
        )
        
        logs = []
        for row in cursor.fetchall():
            logs.append({
                "operation": row[0],
                "status": row[1],
                "error": row[2],
                "timestamp": row[3]
            })
        
        conn.close()
        return logs

import json
import time
import uuid
import requests
import smtplib
from pathlib import Path
from datetime import datetime
from typing import Dict, Any, List, Optional

class PaymentNotificationService:
    """Сервис обработки платежей с уведомлениями."""
    
    def __init__(self, payment_gateway_url: str, merchant_id: str, secret_key: str,
                 smtp_server: str, smtp_port: int, smtp_user: str, smtp_password: str,
                 log_dir: str):
        self.payment_gateway_url = payment_gateway_url
        self.merchant_id = merchant_id
        self.secret_key = secret_key
        self.smtp_server = smtp_server
        self.smtp_port = smtp_port
        self.smtp_user = smtp_user
        self.smtp_password = smtp_password
        self.log_dir = Path(log_dir)
        self.log_dir.mkdir(parents=True, exist_ok=True)
        self.session = requests.Session()
    
    def _log_transaction(self, data: Dict[str, Any]) -> None:
        """Логирование транзакции в файл."""
        timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
        log_file = self.log_dir / f"payment_{timestamp}.json"
        with open(log_file, 'w') as f:
            json.dump({"timestamp": datetime.now().isoformat(), **data}, f)
    
    def _send_notification(self, recipient: str, subject: str, body: str) -> None:
        """Отправка уведомления по email."""
        msg = MIMEMultipart()
        msg['From'] = self.smtp_user
        msg['To'] = recipient
        msg['Subject'] = subject
        msg.attach(MIMEText(body, 'plain'))
        
        with smtplib.SMTP(self.smtp_server, self.smtp_port) as server:
            server.starttls()
            server.login(self.smtp_user, self.smtp_password)
            server.send_message(msg)
    
    def process_payment(self, amount: float, currency: str, card_token: str,
                        customer_email: str, description: str) -> Dict[str, Any]:
        """Обработка платежа с уведомлением."""
        # 1. Обработка платежа через шлюз
        payment_url = f"{self.payment_gateway_url}/api/v1/charge"
        payment_data = {
            "merchant_id": self.merchant_id,
            "amount": amount,
            "currency": currency,
            "card_token": card_token,
            "description": description,

        except requests.exceptions.RequestException as e:

        timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
        filename = f"{prefix}_{timestamp}.json.gz"
        
        # Сжатие
        compressed_data = self._prepare_data(data)

            "cloud_url": cloud_result.get("url"),

# webhook_service.py
import json
import time
import hmac
import hashlib
import requests
from pathlib import Path
from datetime import datetime
from typing import Dict, Any, List, Optional

class WebhookService:
    """Сервис обработки и доставки вебхуков."""
    
    def __init__(self, webhook_url: str, secret_key: str, retry_count: int = 3,
                 retry_delay: int = 5, log_dir: str = "webhooks"):
        self.webhook_url = webhook_url
        self.secret_key = secret_key
        self.retry_count = retry_count
        self.retry_delay = retry_delay
        self.log_dir = Path(log_dir)
        self.log_dir.mkdir(parents=True, exist_ok=True)
        self.session = requests.Session()
    
    def _generate_signature(self, payload: Dict[str, Any]) -> str:
        """Генерация подписи для вебхука."""
        data = json.dumps(payload, sort_keys=True)
        signature = hmac.new(
            self.secret_key.encode(),
            data.encode(),
            hashlib.sha256
        ).hexdigest()
        return signature
    
    def _log_attempt(self, attempt_data: Dict[str, Any]) -> None:
        """Логирование попытки доставки."""
        timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
        log_file = self.log_dir / f"webhook_{timestamp}.json"
        with open(log_file, 'w') as f:
            json.dump(attempt_data, f)
    
    def deliver_webhook(self, payload: Dict[str, Any], event_type: str) -> Dict[str, Any]:
        """Доставка вебхука с повторными попытками."""
        # Генерация подписи
        signature = self._generate_signature(payload)
        
        headers = {
            "Content-Type": "application/json",
            "X-Webhook-Signature": signature,
            "X-Event-Type": event_type
        }
        
        last_error = None
        
        for attempt in range(1, self.retry_count + 1):
            try:
                response = self.session.post(
                    self.webhook_url,
                    json=payload,
                    headers=headers,
                    timeout=5
                )
                
                # Логирование
                self._log_attempt({
                    "attempt": attempt,
                    "url": self.webhook_url,
                    "status_code": response.status_code,
                    "event_type": event_type,
                    "timestamp": datetime.now().isoformat()
                })
                
                if response.status_code in [200, 201, 202]:
                    return {
                        "status": "delivered",
                        "attempts": attempt,
                        "status_code": response.status_code
                    }
                
                # Неуспешный ответ, но не ошибка сети
                if response.status_code >= 400 and response.status_code != 429:
                    # 4xx ошибки не повторяем (кроме 429)
                    if 400 <= response.status_code < 500:
                        raise Exception(f"Client error: {response.status_code}")
                
            except requests.exceptions.Timeout:
                last_error = f"Timeout (attempt {attempt})"
            except requests.exceptions.ConnectionError:
                last_error = f"Connection error (attempt {attempt})"
            except requests.exceptions.HTTPError as e:
                last_error = f"HTTP error: {e}"
            except Exception as e:
                last_error = str(e)
            
            if attempt < self.retry_count:
                time.sleep(self.retry_delay * attempt)
        
        # Все попытки исчерпаны
        return {
            "status": "failed",
            "attempts": self.retry_count,
            "last_error": last_error
        }
    
    def process_webhook_event(self, event: Dict[str, Any]) -> Dict[str, Any]:
        """Обработка одного события вебхука."""
        event_type = event.get("type", "unknown")
        payload = event.get("payload", {})
        
        # Валидация события
        if not payload:
            raise ValueError("Event payload is empty")
        
        # Доставка
        result = self.deliver_webhook(payload, event_type)
        
        return {
            "event_id": event.get("id"),
            "event_type": event_type,
            "delivery_result": result
        }
    
    def process_batch(self, events: List[Dict[str, Any]]) -> List[Dict[str, Any]]:
        """Обработка пакета событий."""
        results = []
        for event in events:
            try:
                results.append(self.process_webhook_event(event))
            except Exception as e:
                results.append({
                    "event_id": event.get("id"),
                    "error": str(e),
                    "status": "processing_failed"
                })
        return results
    
    def get_delivery_logs(self, limit: int = 20) -> List[Dict[str, Any]]:
        """Получение логов доставки."""
        log_files = sorted(
            self.log_dir.glob("webhook_*.json"),
            key=lambda x: x.stat().st_mtime,
            reverse=True
        )[:limit]
        
        logs = []
        for log_file in log_files:
            try:
                with open(log_file, 'r') as f:
                    logs.append(json.load(f))
            except json.JSONDecodeError:
                continue
        
        return logs

    def _get_template(self, template_id: str) -> Dict[str, Any]:
        """Получение шаблона из API или кэша."""
        # Проверка кэша
        cache_file = self.template_cache_dir / f"{template_id}.json"
        if cache_file.exists():
            with open(cache_file, 'r') as f:
                cached = json.load(f)
                # Кэш действителен 1 час
                if time.time() - cached.get("cached_at", 0) < 3600:
                    return cached.get("template", {})
        
        # Загрузка из API
        url = f"{self.template_api_url}/api/v1/templates/{template_id}"
        
        try:
            response = self.session.get(url, timeout=5)
            response.raise_for_status()
            template = response.json()
            
            # Сохранение в кэш
            cache_data = {
                "template": template,
                "cached_at": time.time()
            }
            with open(cache_file, 'w') as f:
                json.dump(cache_data, f)
            
            return template
        except requests.exceptions.Timeout:
            raise Exception("Template API timeout")
        except requests.exceptions.HTTPError as e:
            if e.response.status_code == 404:
                raise Exception(f"Template {template_id} not found")
            raise Exception(f"Template API error: {e}")
        except requests.exceptions.RequestException as e:
            raise Exception(f"Template request failed: {e}")
    
    def _render_template(self, template: Dict[str, Any], data: Dict[str, Any]) -> str:
        """Рендеринг шаблона с данными."""
        content = template.get("content", "")
        for key, value in data.items():
            content = content.replace(f"{{{{{key}}}}}", str(value))
        return content

        # Получение шаблона
        template = self._get_template(template_id)

        subject = self._render_template({"content": template.get("subject", "")}, data)
        body = self._render_template({"content": template.get("content", "")}, data)
        
        # Отправка через SMTP
        msg = MIMEMultipart()
        msg['From'] = self.smtp_user
        msg['To'] = recipient
        msg['Subject'] = subject
        
        if template.get("type") == "html":
            msg.attach(MIMEText(body, 'html'))
        else:
            msg.attach(MIMEText(body, 'plain'))
        
        try:
            with smtplib.SMTP(self.smtp_server, self.smtp_port) as server:
                server.starttls()
                server.login(self.smtp_user, self.smtp_password)
                server.send_message(msg)
            return True
        except smtplib.SMTPAuthenticationError:
            raise Exception("SMTP authentication failed")
        except smtplib.SMTPException as e:
            raise Exception(f"SMTP error: {e}")

            self._last_flush = time.time()

            return False

        except requests.exceptions.RequestException:

    def track_event(self, event_name: str, properties: Dict[str, Any]) -> None:
        """Отслеживание события."""
        event = {
            "event": event_name,
            "properties": properties,
            "timestamp": datetime.now().isoformat(),
            "user_agent": properties.get("user_agent", "unknown"),
            "ip": properties.get("ip", "0.0.0.0")
        }
        
        self._buffer.append(event)
        
        # Принудительный сброс при заполнении буфера
        if len(self._buffer) >= 100:
            self._flush_buffer()
        
        # Проверка интервала сброса
        if time.time() - self._last_flush > self.flush_interval:
            self._flush_buffer()
    
    def track_batch(self, events: List[Dict[str, Any]]) -> None:
        """Отслеживание нескольких событий."""
        for event in events:
            self.track_event(
                event.get("name"),
                event.get("properties", {})
            )
    
    def flush(self) -> None:
        """Принудительный сброс буфера."""
        self._flush_buffer()
    
    def get_event_count(self) -> int:
        """Получение количества событий в буфере."""
        self._load_buffer()
        return len(self._buffer)
    
    def get_stats(self) -> Dict[str, Any]:
        """Получение статистики."""
        self._load_buffer()
        return {
            "buffer_size": len(self._buffer),
            "last_flush": self._last_flush,
            "interval_seconds": self.flush_interval
        }

class TransformationService:
    """Сервис преобразования данных с валидацией через API."""
    
    def __init__(self, api_url: str, api_key: str, input_dir: str, output_dir: str):
        self.api_url = api_url
        self.api_key = api_key
        self.input_dir = Path(input_dir)
        self.output_dir = Path(output_dir)
        self.input_dir.mkdir(parents=True, exist_ok=True)
        self.output_dir.mkdir(parents=True, exist_ok=True)
        self.session = requests.Session()
        self.session.headers.update({
            "Authorization": f"Bearer {api_key}",
            "Content-Type": "application/json"
        })
    
    def _validate_data_via_api(self, data: Dict[str, Any]) -> Dict[str, Any]:
        """Валидация данных через API."""
        url = f"{self.api_url}/api/v1/validate"
        
        try:
            response = self.session.post(url, json={"data": data}, timeout=10)
            response.raise_for_status()
            return response.json()
        except requests.exceptions.Timeout:
            raise Exception("Validation API timeout")
        except requests.exceptions.HTTPError as e:
            if e.response.status_code == 400:
                raise Exception("Invalid data format")
            raise Exception(f"Validation API error: {e}")
        except requests.exceptions.RequestException as e:
            raise Exception(f"Validation request failed: {e}")
    
    def _enrich_data_via_api(self, data: Dict[str, Any]) -> Dict[str, Any]:
        """Обогащение данных через API."""
        url = f"{self.api_url}/api/v1/enrich"
        
        try:
            response = self.session.post(url, json={"data": data}, timeout=10)
            response.raise_for_status()
            return response.json()
        except requests.exceptions.Timeout:
            raise Exception("Enrichment API timeout")
        except requests.exceptions.HTTPError as e:
            raise Exception(f"Enrichment API error: {e}")
        except requests.exceptions.RequestException as e:
            raise Exception(f"Enrichment request failed: {e}")
    
    def transform_json(self, filename: str, output_filename: Optional[str] = None) -> Dict[str, Any]:
        """Преобразование JSON файла."""
        file_path = self.input_dir / filename
        
        if not file_path.exists():

        with open(file_path, 'r') as f:

        enriched_data = self._enrich_data_via_api(data)
        
        # Сохранение результата
        output_filename = output_filename or f"transformed_{datetime.now().strftime('%Y%m%d_%H%M%S')}.json"

                "validation": validation_result,
                "transformed_at": datetime.now().isoformat()
            }, f)
        
        return {
            "output_filename": output_filename,
            "output_path": str(output_path),
            "original_size": len(data),
            "enriched_size": len(enriched_data)
        }

                transformed_data.append(transformed)
            except Exception:

        output_path = self.output_dir / output_filename
        
        with open(output_path, 'w') as f:
            json.dump({
                "original_rows": len(data),
                "transformed_rows": len(transformed_data),
                "data": transformed_data,
                "transformed_at": datetime.now().isoformat()
            }, f)
        
        return {
            "output_filename": output_filename,
            "output_path": str(output_path),
            "total_rows": len(data),
            "transformed_rows": len(transformed_data)
        }

import json

from pathlib import Path
from datetime import datetime
from typing import Dict, Any, List, Optional

class APIMonitorService:
    """Сервис мониторинга API с уведомлениями."""
    
    def __init__(self, monitor_urls: List[str], api_key: str,
                 smtp_server: str, smtp_port: int,
                 smtp_user: str, smtp_password: str,
                 report_dir: str, check_interval: int = 60):
        self.monitor_urls = monitor_urls
        self.api_key = api_key
        self.smtp_server = smtp_server
        self.smtp_port = smtp_port
        self.smtp_user = smtp_user
        self.smtp_password = smtp_password
        self.report_dir = Path(report_dir)
        self.report_dir.mkdir(parents=True, exist_ok=True)
        self.check_interval = check_interval
        self.session = requests.Session()
        self.session.headers.update({
            "Authorization": f"Bearer {api_key}",
            "User-Agent": "APIMonitor/1.0"
        })
        self.status_history = {}
        self.alert_threshold = 3
    
    def _check_url(self, url: str) -> Dict[str, Any]:
        """Проверка одного URL."""
        try:
            start_time = time.time()
            response = self.session.get(url, timeout=5)
            elapsed = time.time() - start_time
            
            return {
                "url": url,
                "status_code": response.status_code,
                "response_time": elapsed,
                "success": response.status_code < 400,
                "timestamp": datetime.now().isoformat()
            }
        except requests.exceptions.Timeout:
            return {
                "url": url,
                "success": False,
                "error": "timeout",
                "timestamp": datetime.now().isoformat()
            }
        except requests.exceptions.ConnectionError:
            return {
                "url": url,
                "success": False,
                "error": "connection_error",
                "timestamp": datetime.now().isoformat()
            }
        except requests.exceptions.RequestException as e:
            return {
                "url": url,
                "success": False,
                "error": str(e),
                "timestamp": datetime.now().isoformat()
            }
    
    def _send_alert(self, url: str, failures: int) -> None:
        """Отправка алерта при сбое."""
        msg = MIMEText(
            f"API {url} has failed {failures} times in a row.\n"
            f"Last check at: {datetime.now()}\n"
            f"Please investigate."
        )
        msg['Subject'] = f"API Alert: {url} is down"
        msg['From'] = self.smtp_user
        msg['To'] = self.smtp_user
        
        try:
            with smtplib.SMTP(self.smtp_server, self.smtp_port) as server:
                server.starttls()
                server.login(self.smtp_user, self.smtp_password)
                server.send_message(msg)
        except Exception:
            pass

        """Сохранение отчёта в файл."""

        report_file = self.report_dir / f"monitor_{timestamp}.json"
        
        with open(report_file, 'w') as f:
            json.dump({
                "timestamp": timestamp,
                "results": results
            }, f)
    
    def check_all(self) -> List[Dict[str, Any]]:
        """Проверка всех URL."""
        results = []
        for url in self.monitor_urls:
            result = self._check_url(url)
            results.append(result)
            
            # Обновление статуса
            if result["success"]:
                self.status_history[url] = 0
            else:
                self.status_history[url] = self.status_history.get(url, 0) + 1
                if self.status_history[url] >= self.alert_threshold:
                    self._send_alert(url, self.status_history[url])
        
        self._save_report(results)
        return results
    
    def run_continuously(self, duration: Optional[int] = None) -> List[List[Dict[str, Any]]]:
        """Запуск мониторинга в цикле."""
        all_reports = []
        start_time = time.time()
        
        while True:
            if duration and (time.time() - start_time) > duration:
                break
            
            results = self.check_all()
            all_reports.append(results)
            
            time.sleep(self.check_interval)
        
        return all_reports
    
    def get_summary(self) -> Dict[str, Any]:
        """Получение сводки статуса."""
        summary = {}
        for url in self.monitor_urls:
            if url in self.status_history:
                summary[url] = {
                    "consecutive_failures": self.status_history[url],
                    "status": "DOWN" if self.status_history[url] >= self.alert_threshold else "UP"
                }
            else:
                summary[url] = {
                    "consecutive_failures": 0,
                    "status": "UNKNOWN"
                }
        return summary
