## 1. Введение: От простых моков к сложным

В предыдущей лекции мы рассмотрели базовые техники мокирования внешних зависимостей. Однако в реальных проектах часто возникает необходимость **проверять не только возвращаемое значение**, но и:

- **С какими аргументами** был вызван метод.
- **Сколько раз** он был вызван.
- **В каком порядке** происходили вызовы.
- **Какие методы** были вызваны на мок-объекте.

Это называется **верификацией взаимодействий (interaction verification)** — ключевой аспект тестирования на основе поведения (Behavior-Driven Testing).

## 2. Базовые методы проверки вызовов

### 2.1. `assert_called()`, `assert_not_called()`, `assert_called_once()`

```python
from unittest.mock import Mock

def test_basic_call_assertions():
    mock = Mock()
    
    # Проверка, что мок НЕ вызывался
    mock.assert_not_called()
    
    # Вызываем метод
    mock.process(10)
    
    # Проверка, что вызывался хотя бы раз
    mock.assert_called()
    
    # Проверка, что вызывался ровно один раз
    mock.assert_called_once()
    
    # Ещё один вызов — теперь assert_called_once() упадёт
    mock.process(20)
    # mock.assert_called_once()  # AssertionError: Expected 'mock' to have been called once.

4.2. Моки с настраиваемым поведением (side_effect)
side_effect может быть:

Функцией — вызывается при каждом вызове мока.

Исключением — выбрасывается при вызове.

Итерацией — последовательность возвращаемых значений.

python
def test_side_effect_function():
    mock = Mock()
    
    def process_call(*args, **kwargs):
        if args[0] < 0:
            raise ValueError("Negative value not allowed")
        return args[0] * 2
    
    mock.process.side_effect = process_call
    
    assert mock.process(5) == 10
    assert mock.process(10) == 20
    
    with pytest.raises(ValueError, match="Negative value not allowed"):
        mock.process(-1)
    
    # Проверка, что функция вызывалась с правильными аргументами
    assert mock.process.call_args_list == [
        ((5,), {}),
        ((10,), {}),
        ((-1,), {}),
    ]

def test_side_effect_exception():
    mock = Mock()
    mock.method.side_effect = RuntimeError("Something went wrong")
    
    with pytest.raises(RuntimeError, match="Something went wrong"):
        mock.method()

def test_side_effect_iterable():
    mock = Mock()
    mock.method.side_effect = [10, 20, 30]
    
    assert mock.method() == 10
    assert mock.method() == 20
    assert mock.method() == 30

from unittest.mock import call

def test_assert_has_calls():

6. Реальные сценарии использования
6.1. Сценарий: Тестирование обработчика заказа
python
class OrderProcessor:
    def __init__(self, payment_gateway, email_sender):
        self.payment_gateway = payment_gateway
        self.email_sender = email_sender
    
    def process_order(self, order):
        # 1. Проверка заказа
        if not order.get('items'):
            raise ValueError("Order has no items")
        
        # 2. Обработка платежа
        payment_result = self.payment_gateway.charge(order['total'])
        if not payment_result.get('success'):
            return {"status": "failed", "reason": payment_result.get('message')}
        
        # 3. Отправка подтверждения
        self.email_sender.send(
            order['email'],
            "Order confirmed",
            f"Your order #{order['id']} has been confirmed."
        )
        
        return {"status": "success", "payment_id": payment_result['payment_id']}
Тесты:

python
from unittest.mock import Mock, call

def test_process_order_success():
    # Создаём моки
    gateway = Mock()
    email = Mock()
    processor = OrderProcessor(gateway, email)
    
    # Настраиваем поведение
    gateway.charge.return_value = {
        "success": True,
        "payment_id": "pay_123"
    }
    
    order = {
        "id": 1001,
        "email": "user@example.com",
        "total": 150.0,
        "items": [{"name": "product", "price": 150.0}]
    }
    
    # Выполняем
    result = processor.process_order(order)
    
    # Проверяем взаимодействия
    gateway.charge.assert_called_once_with(150.0)
    email.send.assert_called_once_with(
        "user@example.com",
        "Order confirmed",
        "Your order #1001 has been confirmed."
    )
    
    assert result["status"] == "success"
    assert result["payment_id"] == "pay_123"

def test_process_order_payment_failed():
    gateway = Mock()
    email = Mock()
    processor = OrderProcessor(gateway, email)
    
    gateway.charge.return_value = {
        "success": False,
        "message": "Insufficient funds"
    }
    
    order = {
        "id": 1002,
        "email": "user@example.com",
        "total": 150.0,
        "items": [{"name": "product", "price": 150.0}]
    }
    
    result = processor.process_order(order)
    
    # Проверяем, что email не отправлялся
    email.send.assert_not_called()
    assert result["status"] == "failed"
    assert result["reason"] == "Insufficient funds"

def test_process_order_validation_error():
    gateway = Mock()

6.2. Сценарий: Тестирование кэширующего сервиса
python
class CachingService:
    def __init__(self, cache_backend, data_source):
        self.cache = cache_backend
        self.data_source = data_source
    
    def get_data(self, key):
        # Проверяем кэш
        cached = self.cache.get(key)
        if cached is not None:
            return cached
        
        # Загружаем из источника
        data = self.data_source.fetch(key)
        
        # Сохраняем в кэш
        self.cache.set(key, data)
        return data
    
    def invalidate(self, key):
        self.cache.delete(key)
Тесты:

python
def test_caching_service_cache_hit():
    cache = Mock()
    source = Mock()
    service = CachingService(cache, source)
    
    # Настраиваем поведение кэша
    cache.get.return_value = {"data": "cached_value"}
    
    result = service.get_data("user_123")
    
    assert result == {"data": "cached_value"}
    
    # Проверяем, что данные из источника не запрашивались
    source.fetch.assert_not_called()
    cache.set.assert_not_called()
    
    # Проверяем, что кэш был вызван с правильным ключом
    cache.get.assert_called_once_with("user_123")

def test_caching_service_cache_miss():
    cache = Mock()

    service = CachingService(cache, source)
    
    # Кэш не содержит данных

    source.fetch.return_value = {"data": "fresh_data"}

    # Проверяем последовательность вызовов

    source.fetch.assert_called_once_with("user_123")
    cache.set.assert_called_once_with("user_123", {"data": "fresh_data"})
    
    # Проверяем точный порядок
    expected_calls = [call.get("user_123"), call.set("user_123", {"data": "fresh_data"})]
    # Для проверки порядка можно использовать assert_has_calls

def test_caching_service_invalidate():
    cache = Mock()

8. Частые ошибки при мокировании
Ошибка	Описание	Решение
Мокирование не того объекта	Импорт в коде отличается от импорта в тесте	Мокируйте объект там, где он используется, а не где определён
Неверный путь для patch	patch('my_module.SomeClass') вместо patch('other_module.SomeClass')	Проверьте импорты
Изменение мока между тестами	Один мок меняет поведение для следующих тестов	Используйте изолированные моки в каждом тесте
Неполная проверка	Проверяете только return_value, но не аргументы	Используйте assert_called_with
Игнорирование вложенных моков	Проверяете только поверхностные вызовы	Проверяйте всю цепочку вызовов
Слишком много моков	Сложно поддерживать и понимать тесты	Мокируйте только необходимое
9. Итоговый вывод по лекции
Создание сложных моков и проверка вызовов — это ключевой навык для тестирования взаимодействий между компонентами.

Ключевые тезисы для запоминания:

Проверка вызовов: assert_called_once(), assert_called_with(), assert_has_calls(), assert_any_call().

Сложные моки: вложенные моки, side_effect (функции, исключения, итераторы), spec и configure_mock().

Интроспекция вызовов: call_args, call_args_list, call_count, method_calls.

Порядок вызовов: assert_has_calls() и ANY для частичного сопоставления.

Специальные техники: pytest-mock (mocker.patch, mocker.spy), AsyncMock.

Задание: Разработать тесты для класса NotificationService, который отправляет уведомления через разные каналы (email, SMS, push).

Код для тестирования:

python
class NotificationService:
    def __init__(self, email_sender, sms_sender, push_sender):
        self.email = email_sender
        self.sms = sms_sender
        self.push = push_sender
    
    def send_notification(self, user, message, channels=None):
        if channels is None:
            channels = ["email", "sms", "push"]
        
        results = {}
        
        if "email" in channels and user.get("email"):
            results["email"] = self.email.send(user["email"], message)
        
        if "sms" in channels and user.get("phone"):
            results["sms"] = self.sms.send(user["phone"], message)
        
        if "push" in channels and user.get("device_id"):
            results["push"] = self.push.send(user["device_id"], message)
        
        # Агрегация результатов
        failed = [k for k, v in results.items() if not v.get("success", False)]
        
        if failed:
            raise Exception(f"Failed channels: {', '.join(failed)}")
        
        return results
