## 1. Введение: Почему API-клиенты нужно тестировать с моками

API-клиенты — это код, который отправляет HTTP-запросы к внешним сервисам. Тестирование таких клиентов без моков приводит к серьёзным проблемам:

| Проблема | Описание |
|----------|----------|
| **Медленные тесты** | Каждый тест ждёт реального ответа от сервера (200-500 мс) |
| **Ненадёжные тесты** | Тесты падают из-за проблем с сетью, сервером или квотами API |
| **Побочные эффекты** | Тесты могут создавать реальные данные, отправлять email, списывать деньги |
| **Невозможность тестировать ошибки** | Нельзя гарантировать, что сервер вернёт ошибку 500 или тайм-аут |
| **Зависимость от среды** | Тесты работают только при наличии доступа к API |

**Решение:** Заменять реальные HTTP-запросы на **моки (mocks)** — объекты-заглушки, которые имитируют ответы API.

import requests
from requests.exceptions import RequestException

class UserAPI:
    BASE_URL = "https://api.example.com"
    
    def __init__(self, api_key, timeout=10):
        self.api_key = api_key
        self.timeout = timeout
        self.session = requests.Session()
        self.session.headers.update({
            "Authorization": f"Bearer {api_key}",
            "Content-Type": "application/json"
        })
    
    def get_user(self, user_id):
        """Получение пользователя по ID."""
        url = f"{self.BASE_URL}/users/{user_id}"
        try:
            response = self.session.get(url, timeout=self.timeout)
            response.raise_for_status()
            return response.json()
        except requests.exceptions.Timeout:
            raise UserAPIError("Timeout while fetching user")
        except requests.exceptions.HTTPError as e:
            if e.response.status_code == 404:
                raise UserNotFoundError(f"User {user_id} not found")
            raise UserAPIError(f"HTTP error: {e}")
        except RequestException as e:
            raise UserAPIError(f"Request failed: {e}")
    
    def create_user(self, name, email):
        """Создание нового пользователя."""
        url = f"{self.BASE_URL}/users"
        data = {"name": name, "email": email}
        try:
            response = self.session.post(url, json=data, timeout=self.timeout)
            response.raise_for_status()
            return response.json()
        except requests.exceptions.Timeout:
            raise UserAPIError("Timeout while creating user")
        except requests.exceptions.HTTPError as e:
            if e.response.status_code == 400:
                raise ValidationError("Invalid user data")
            raise UserAPIError(f"HTTP error: {e}")
        except RequestException as e:
            raise UserAPIError(f"Request failed: {e}")
    
    def delete_user(self, user_id):
        """Удаление пользователя."""
        url = f"{self.BASE_URL}/users/{user_id}"
        try:
            response = self.session.delete(url, timeout=self.timeout)
            response.raise_for_status()
            return True
        except requests.exceptions.HTTPError as e:
            if e.response.status_code == 404:
                raise UserNotFoundError(f"User {user_id} not found")
            raise UserAPIError(f"HTTP error: {e}")
        except RequestException as e:
            raise UserAPIError(f"Request failed: {e}")

class UserAPIError(Exception):
    pass

class UserNotFoundError(UserAPIError):
    pass

class ValidationError(UserAPIError):
    pass

import httpx
from httpx import HTTPError, TimeoutException

class AsyncUserAPI:
    BASE_URL = "https://api.example.com"
    
    def __init__(self, api_key, timeout=10):
        self.api_key = api_key
        self.timeout = timeout
        self.headers = {
            "Authorization": f"Bearer {api_key}",
            "Content-Type": "application/json"
        }
    
    async def get_user(self, user_id):
        url = f"{self.BASE_URL}/users/{user_id}"
        async with httpx.AsyncClient() as client:
            try:
                response = await client.get(url, headers=self.headers, timeout=self.timeout)
                response.raise_for_status()
                return response.json()
            except TimeoutException:
                raise UserAPIError("Timeout while fetching user")
            except HTTPError as e:
                if e.response and e.response.status_code == 404:
                    raise UserNotFoundError(f"User {user_id} not found")
                raise UserAPIError(f"HTTP error: {e}")

8. Итоговый вывод по лекции
Тестирование API-клиентов с моками — это обязательная практика для обеспечения надёжности интеграций с внешними сервисами.

Ключевые тезисы для запоминания:

Мокируйте HTTP-запросы — никогда не обращайтесь к реальным API в модульных тестах.

Используйте специализированные инструменты: responses для requests, pytest_httpx для httpx.

Проверяйте все сценарии: успех, ошибки 4xx/5xx, тайм-ауты.

Верифицируйте запросы: проверяйте URL, метод, заголовки, тело.

Используйте фикстуры для переиспользования клиента и моков.

Тестируйте обработку ошибок — она не менее важна, чем успешные сценарии.

import requests
from requests.exceptions import RequestException

class WeatherAPI:
    BASE_URL = "https://api.weather.com"
    
    def __init__(self, api_key):
        self.api_key = api_key
        self.session = requests.Session()
        self.session.params = {"apikey": api_key}
    
    def get_current_weather(self, city):
        url = f"{self.BASE_URL}/v1/current.json"
        try:
            response = self.session.get(url, params={"q": city}, timeout=5)
            response.raise_for_status()
            data = response.json()
            return {
                "city": data["location"]["name"],
                "temperature": data["current"]["temp_c"],
                "condition": data["current"]["condition"]["text"]
            }
        except requests.exceptions.Timeout:
            raise WeatherAPIError("Timeout getting weather")
        except requests.exceptions.HTTPError as e:
            if e.response.status_code == 401:
                raise WeatherAPIError("Invalid API key")
            elif e.response.status_code == 404:
                raise CityNotFoundError(f"City {city} not found")
            raise WeatherAPIError(f"HTTP error: {e}")
        except RequestException as e:
            raise WeatherAPIError(f"Request failed: {e}")
    
    def get_forecast(self, city, days=3):
        url = f"{self.BASE_URL}/v1/forecast.json"
        try:
            response = self.session.get(url, params={"q": city, "days": days}, timeout=5)
            response.raise_for_status()
            data = response.json()
            return data["forecast"]["forecastday"]
        except requests.exceptions.Timeout:
            raise WeatherAPIError("Timeout getting forecast")
        except RequestException as e:
            raise WeatherAPIError(f"Request failed: {e}")

class WeatherAPIError(Exception):
    pass

class CityNotFoundError(WeatherAPIError):
    pass
