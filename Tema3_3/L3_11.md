## 1. Введение: Почему кастомные исключения важны для тестирования

Стандартные исключения Python (например, `ValueError`, `TypeError`, `RuntimeError`) — это универсальные инструменты, которые несут ограниченную информацию об ошибке. В сложных приложениях этого недостаточно.

**Основные причины использовать кастомные исключения:**

| Причина | Описание |
|---------|----------|
| **Семантическая точность** | Исключение должно отражать конкретную ошибку в предметной области |
| **Обработка по типам** | Клиентский код может обрабатывать разные ошибки по-разному |
| **Дополнительная информация** | Кастомные поля (код ошибки, поле, значение, причина) |
| **Иерархия** | Группировка родственных ошибок, обработка на разных уровнях |
| **Тестируемость** | Легче изолировать и проверять конкретные сценарии |

**Пример:**
- `ValueError("Invalid email")` — общее сообщение.
- `EmailValidationError("Invalid email format", field="email", value="test")` — конкретное, с контекстом.

class ValidationError(Exception):
    def __init__(self, message, field, value, code=None):
        super().__init__(message)
        self.field = field
        self.value = value
        self.code = code or 400
    
    def __str__(self):
        return f"[{self.code}] {self.field}: {self.message} (value: {self.value})"

3.3. Реализация иерархии
python
# Базовое исключение приложения
class ApplicationException(Exception):
    """Базовое исключение для всего приложения."""
    def __init__(self, message, code=500):
        self.message = message
        self.code = code
        super().__init__(message)

# Доменные исключения (бизнес-логика)
class DomainException(ApplicationException):
    """Ошибка бизнес-логики."""
    def __init__(self, message, code=400, field=None):
        super().__init__(message, code)
        self.field = field

class OrderException(DomainException):
    def __init__(self, message, code=400, order_id=None):
        super().__init__(message, code)
        self.order_id = order_id

class OrderNotFoundException(OrderException):
    def __init__(self, order_id):
        super().__init__(
            f"Order with id {order_id} not found",
            code=404,
            order_id=order_id
        )

class OrderValidationException(OrderException):
    def __init__(self, message, field, value):
        super().__init__(f"Order validation failed: {message}", code=400)
        self.field = field
        self.value = value

# Инфраструктурные исключения
class InfrastructureException(ApplicationException):
    def __init__(self, message, code=500, component=None):
        super().__init__(message, code)
        self.component = component

class DatabaseException(InfrastructureException):
    def __init__(self, message, query=None):
        super().__init__(
            f"Database error: {message}",
            code=500,
            component="DATABASE"
        )
        self.query = query

class ConnectionPoolExhaustedException(DatabaseException):
    def __init__(self):
        super().__init__("Connection pool exhausted", query=None)

def get_user_from_db(user_id):
    try:
        # Запрос к БД...
        result = db.query("SELECT * FROM users WHERE id = ?", user_id)
        if not result:
            raise DatabaseException("User not found in database")
        return result
    except DatabaseException as e:
        # Преобразование в доменное исключение
        raise UserNotFoundException(user_id) from e

5.2. Агрегация ошибок
Идея: Собирать несколько ошибок в одно исключение для валидации формы.

python
class ValidationAggregateError(Exception):
    def __init__(self, errors):
        self.errors = errors
        messages = [f"{e.field}: {e.message}" for e in errors]
        super().__init__("\n".join(messages))

def validate_user_data(data):
    errors = []
    
    if not data.get("email"):
        errors.append(ValidationError("Email is required", "email", data.get("email")))
    elif "@" not in data["email"]:
        errors.append(ValidationError("Invalid email format", "email", data["email"]))
    
    if not data.get("age"):
        errors.append(ValidationError("Age is required", "age", data.get("age")))
    elif data["age"] < 18:
        errors.append(ValidationError("Age must be >= 18", "age", data["age"]))
    
    if errors:
        raise ValidationAggregateError(errors)
Тест:

python
def test_validation_aggregation():
    with pytest.raises(ValidationAggregateError) as exc_info:
        validate_user_data({"email": "invalid", "age": 15})
    
    assert len(exc_info.value.errors) == 2
    assert any(e.field == "email" for e in exc_info.value.errors)
    assert any(e.field == "age" for e in exc_info.value.errors)
    assert "invalid email" in str(exc_info.value).lower()

class PaymentProcessor:
    def process_payment(self, amount, method, account_id):
        if amount <= 0:
            raise PaymentAmountException(amount)
        
        if method not in ["card", "paypal"]:
            raise PaymentMethodException(method)
        
        try:
            # Внешний вызов...
            result = external_service.charge(account_id, amount)
        except NetworkError:
            raise PaymentServiceException("Network error", cause="network")
        except TimeoutError:
            raise PaymentServiceException("Service timeout", cause="timeout")
        
        if not result.success:
            raise PaymentDeclinedException(result.reason)
        
        return result

7. Сравнение подходов к исключениям
Подход	Преимущества	Недостатки
Стандартные исключения	Простота, стандартизация	Недостаточно контекста
Кастомные исключения	Контекст, семантика	Больше кода, иерархия
Результат (Result type)	Явные возвращаемые значения	Громоздко, нестандартно
Either/Option	Функциональный стиль	Не pythonic
8. Итоговый вывод по лекции
Кастомные исключения и иерархия исключений — это не просто "хороший стиль", а инструмент для построения надёжного и тестируемого приложения.

Ключевые тезисы для запоминания:

Кастомные исключения содержат семантическую и контекстную информацию.

Иерархия исключений позволяет группировать ошибки и обрабатывать их на нужном уровне абстракции.

При тестировании проверяйте не только тип, но и атрибуты, сообщения и причину.

Используйте __cause__ для сохранения контекста при преобразовании исключений.

Агрегация ошибок полезна для валидации форм и панелей.

Покрывайте все ветви обработки ошибок в тестах.

Логирование ошибок — часть обработки, проверяйте его в тестах.

Код для тестирования:

python
class TaskError(Exception):
    pass

class TaskValidationError(TaskError):
    def __init__(self, message, field, value):
        super().__init__(message)
        self.field = field
        self.value = value

class TaskNotFoundError(TaskError):
    def __init__(self, task_id):
        super().__init__(f"Task {task_id} not found")
        self.task_id = task_id

class TaskStatusError(TaskError):
    def __init__(self, task_id, current_status, attempted_status):
        super().__init__(
            f"Cannot transition from {current_status} to {attempted_status}"
        )
        self.task_id = task_id
        self.current_status = current_status
        self.attempted_status = attempted_status

class TaskManager:
    def __init__(self):
        self.tasks = {}
        self.next_id = 1
    
    def create_task(self, title, description=""):
        if not title or len(title) < 3:
            raise TaskValidationError("Title too short", "title", title)
        task_id = self.next_id
        self.tasks[task_id] = {"id": task_id, "title": title, "status": "NEW"}
        self.next_id += 1
        return task_id
    
    def get_task(self, task_id):
        if task_id not in self.tasks:
            raise TaskNotFoundError(task_id)
        return self.tasks[task_id]
    
    def update_status(self, task_id, new_status):
        task = self.get_task(task_id)
        if task["status"] not in ["NEW", "IN_PROGRESS", "DONE"]:
            raise TaskStatusError(task_id, task["status"], new_status)
        if new_status == task["status"]:
            return
        allowed = {
            "NEW": ["IN_PROGRESS"],
            "IN_PROGRESS": ["DONE", "NEW"],
            "DONE": ["NEW"]
        }
        if new_status not in allowed.get(task["status"], []):
            raise TaskStatusError(task_id, task["status"], new_status)
        task["status"] = new_status
    
    def delete_task(self, task_id):
        if task_id not in self.tasks:
            raise TaskNotFoundError(task_id)
        del self.tasks[task_id]
