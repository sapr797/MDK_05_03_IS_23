| Аспект | Что проверяем |
|--------|---------------|

# order_system.py
import logging
from datetime import datetime

logger = logging.getLogger(__name__)

class OrderError(Exception):
    def __init__(self, message, code=500, details=None):
        self.message = message
        self.code = code
        self.details = details or {}
        super().__init__(message)

class OrderNotFoundError(OrderError):
    def __init__(self, order_id):
        super().__init__(f"Order {order_id} not found", code=404)
        self.order_id = order_id

class OrderValidationError(OrderError):
    def __init__(self, message, field, value, details=None):
        super().__init__(message, code=400, details=details)
        self.field = field
        self.value = value

class OrderStatusError(OrderError):
    def __init__(self, order_id, current_status, attempted_status):
        super().__init__(
            f"Cannot transition from {current_status} to {attempted_status}",
            code=400
        )
        self.order_id = order_id
        self.current_status = current_status
        self.attempted_status = attempted_status

class OrderSystem:
    def __init__(self):
        self.orders = {}
        self.next_id = 1
    
    def create_order(self, user_id, items, delivery_address):
        try:
            # Валидация
            if not user_id or user_id <= 0:
                raise OrderValidationError("Invalid user ID", "user_id", user_id)
            
            if not items:
                raise OrderValidationError("Order must have at least one item", "items", items)
            
            if not delivery_address or len(delivery_address) < 5:
                raise OrderValidationError("Delivery address too short", "delivery_address", delivery_address)
            
            # Проверка каждого товара
            for item in items:
                if item.get("quantity", 0) <= 0:
                    raise OrderValidationError(
                        "Quantity must be positive",
                        "quantity",
                        item.get("quantity"),
                        details={"item": item.get("name", "unknown")}
                    )
                if item.get("price", 0) <= 0:
                    raise OrderValidationError(
                        "Price must be positive",
                        "price",
                        item.get("price"),
                        details={"item": item.get("name", "unknown")}
                    )
            
            order_id = self.next_id
            total = sum(item["price"] * item["quantity"] for item in items)
            self.orders[order_id] = {
                "id": order_id,
                "user_id": user_id,
                "items": items,
                "delivery_address": delivery_address,
                "total": total,
                "status": "PENDING",
                "created_at": datetime.now()
            }
            self.next_id += 1
            return order_id
        
        except OrderValidationError as e:
            logger.error(f"Validation error: {e.field}={e.value} - {e.message}")
            raise
        except Exception as e:
            logger.error(f"Unexpected error: {e}")
            raise OrderError("Failed to create order", code=500, details={"original_error": str(e)}) from e
    
    def update_order_status(self, order_id, new_status):
        try:
            order = self.orders.get(order_id)
            if not order:
                raise OrderNotFoundError(order_id)
            
            allowed_transitions = {
                "PENDING": ["CONFIRMED", "CANCELLED"],
                "CONFIRMED": ["SHIPPED", "CANCELLED"],
                "SHIPPED": ["DELIVERED"],
                "DELIVERED": [],
                "CANCELLED": []
            }
            
            if new_status not in allowed_transitions.get(order["status"], []):
                raise OrderStatusError(order_id, order["status"], new_status)
            
            old_status = order["status"]
            order["status"] = new_status
            logger.info(f"Order {order_id} status changed from {old_status} to {new_status}")
            return order
        
        except OrderNotFoundError as e:
            logger.error(f"Order not found: {e.order_id}")
            raise
        except OrderStatusError as e:
            logger.error(f"Status transition error: {e}")
            raise
        except Exception as e:
            logger.error(f"Unexpected error updating order {order_id}: {e}")
            raise OrderError("Failed to update order status", code=500) from e
    
    def cancel_order(self, order_id, reason):
        try:
            order = self.orders.get(order_id)
            if not order:
                raise OrderNotFoundError(order_id)
            
            if order["status"] in ["SHIPPED", "DELIVERED"]:
                raise OrderStatusError(order_id, order["status"], "CANCELLED")
            
            order["status"] = "CANCELLED"
            order["cancellation_reason"] = reason
            logger.info(f"Order {order_id} cancelled: {reason}")
            return True
        
        except OrderNotFoundError as e:
            logger.error(f"Order not found: {e.order_id}")
            raise
        except OrderStatusError as e:
            logger.error(f"Cannot cancel order {e.order_id}: {e}")
            raise
        except Exception as e:
            logger.error(f"Unexpected error cancelling order {order_id}: {e}")
            raise OrderError("Failed to cancel order", code=500) from e
    
    def get_order(self, order_id):
        order = self.orders.get(order_id)
        if not order:
            raise OrderNotFoundError(order_id)
        return order

# payment_system.py
import logging
from datetime import datetime

logger = logging.getLogger(__name__)

class PaymentError(Exception):
    def __init__(self, message, code=500, payment_id=None, details=None):
        self.message = message
        self.code = code
        self.payment_id = payment_id
        self.details = details or {}
        super().__init__(message)

class PaymentValidationError(PaymentError):
    def __init__(self, message, field, value, payment_id=None):
        super().__init__(message, code=400, payment_id=payment_id)
        self.field = field
        self.value = value

class PaymentProcessingError(PaymentError):
    def __init__(self, payment_id, reason, details=None):
        super().__init__(
            f"Payment {payment_id} processing failed: {reason}",
            code=402,
            payment_id=payment_id,
            details=details
        )
        self.reason = reason

class PaymentRefundError(PaymentError):
    def __init__(self, payment_id, reason, details=None):
        super().__init__(
            f"Refund for payment {payment_id} failed: {reason}",
            code=409,
            payment_id=payment_id,
            details=details
        )
        self.reason = reason

class PaymentSystem:
    def __init__(self):
        self.payments = {}
        self.next_id = 1
    
    def process_payment(self, order_id, amount, method, card_data=None):
        try:
            # Валидация
            if not order_id:
                raise PaymentValidationError("Order ID is required", "order_id", order_id)
            
            if amount <= 0:
                raise PaymentValidationError("Amount must be positive", "amount", amount)
            
            if method not in ["CARD", "PAYPAL", "WALLET"]:
                raise PaymentValidationError("Invalid payment method", "method", method)
            
            if method == "CARD":
                if not card_data or not card_data.get("number"):
                    raise PaymentValidationError("Card number is required", "card_number", None)
                if len(card_data.get("number", "")) < 16:
                    raise PaymentValidationError("Invalid card number", "card_number", card_data.get("number"))
                if card_data.get("expiry", "") < datetime.now().strftime("%Y%m"):
                    raise PaymentValidationError("Card has expired", "expiry", card_data.get("expiry"))
            
            # Имитация обработки платежа
            # 50% шанс ошибки для тестирования
            import random
            if random.random() < 0.3:
                raise PaymentProcessingError(
                    self.next_id,
                    "Payment gateway unavailable",
                    details={"gateway": "stripe", "attempt": 1}
                )
            
            payment_id = self.next_id
            self.payments[payment_id] = {
                "id": payment_id,
                "order_id": order_id,
                "amount": amount,
                "method": method,
                "status": "SUCCESS",
                "processed_at": datetime.now(),
                "transaction_id": f"txn_{payment_id}_{order_id}"
            }
            self.next_id += 1
            logger.info(f"Payment {payment_id} processed successfully for order {order_id}")
            return payment_id
        
        except PaymentValidationError as e:
            logger.error(f"Payment validation error: {e.field}={e.value} - {e.message}")
            raise
        except PaymentProcessingError as e:
            logger.error(f"Payment processing error: {e}")
            raise
        except Exception as e:
            logger.error(f"Unexpected error: {e}")
            raise PaymentError("Failed to process payment", code=500) from e
    
    def refund_payment(self, payment_id, reason="Customer request"):
        try:
            payment = self.payments.get(payment_id)
            if not payment:
                raise PaymentError(f"Payment {payment_id} not found", code=404, payment_id=payment_id)
            
            if payment["status"] != "SUCCESS":
                raise PaymentRefundError(payment_id, f"Cannot refund payment with status {payment['status']}")
            
            # Имитация возврата
            import random
            if random.random() < 0.2:
                raise PaymentRefundError(
                    payment_id,
                    "Refund gateway error",
                    details={"gateway": "stripe", "refund_id": f"ref_{payment_id}"}
                )
            
            payment["status"] = "REFUNDED"
            payment["refund_reason"] = reason
            payment["refunded_at"] = datetime.now()
            logger.info(f"Payment {payment_id} refunded: {reason}")
            return True
        
        except PaymentRefundError as e:
            logger.error(f"Refund error: {e}")
            raise
        except Exception as e:
            logger.error(f"Unexpected error refunding payment {payment_id}: {e}")
            raise PaymentError("Failed to process refund", code=500, payment_id=payment_id) from e
    
    def get_payment(self, payment_id):
        payment = self.payments.get(payment_id)
        if not payment:
            raise PaymentError(f"Payment {payment_id} not found", code=404, payment_id=payment_id)
        return payment

# user_system.py
import logging
import re
from datetime import datetime

logger = logging.getLogger(__name__)

class UserError(Exception):
    def __init__(self, message, code=500, user_id=None, details=None):
        self.message = message
        self.code = code
        self.user_id = user_id
        self.details = details or {}
        super().__init__(message)

class UserNotFoundError(UserError):
    def __init__(self, user_id):
        super().__init__(f"User {user_id} not found", code=404, user_id=user_id)

class UserValidationError(UserError):
    def __init__(self, message, field, value, user_id=None):
        super().__init__(message, code=400, user_id=user_id)
        self.field = field
        self.value = value

class UserEmailError(UserError):
    def __init__(self, email, reason, user_id=None):
        super().__init__(f"Email {email} error: {reason}", code=400, user_id=user_id)
        self.email = email
        self.reason = reason

class UserAuthorizationError(UserError):
    def __init__(self, user_id, action, reason):
        super().__init__(
            f"User {user_id} not authorized for {action}: {reason}",
            code=403,
            user_id=user_id
        )
        self.action = action
        self.reason = reason

class UserSystem:
    def __init__(self):
        self.users = {}
        self.next_id = 1
    
    def create_user(self, name, email, password, role="USER"):
        try:
            # Валидация
            if not name or len(name) < 2:
                raise UserValidationError("Name must be at least 2 characters", "name", name)
            
            if not email or "@" not in email:
                raise UserValidationError("Invalid email format", "email", email)
            
            if not password or len(password) < 8:
                raise UserValidationError("Password must be at least 8 characters", "password", "***")
            
            if not re.search(r"[A-Z]", password):
                raise UserValidationError("Password must contain uppercase letter", "password", "***")
            
            if not re.search(r"[a-z]", password):
                raise UserValidationError("Password must contain lowercase letter", "password", "***")
            
            if not re.search(r"\d", password):
                raise UserValidationError("Password must contain digit", "password", "***")
            
            if any(u["email"] == email for u in self.users.values()):
                raise UserEmailError(email, "Email already registered")
            
            user_id = self.next_id
            self.users[user_id] = {
                "id": user_id,
                "name": name,
                "email": email,
                "role": role,
                "active": True,
                "created_at": datetime.now()
            }
            self.next_id += 1
            logger.info(f"User {user_id} created with email {email}")
            return user_id
        
        except (UserValidationError, UserEmailError) as e:
            logger.error(f"User creation failed: {e}")
            raise
        except Exception as e:
            logger.error(f"Unexpected error creating user: {e}")
            raise UserError("Failed to create user", code=500) from e
    
    def get_user(self, user_id):
        user = self.users.get(user_id)
        if not user:
            raise UserNotFoundError(user_id)
        return user
    
    def deactivate_user(self, user_id, admin_id=None):
        try:
            user = self.get_user(user_id)
            
            # Проверка прав
            if admin_id:
                admin = self.get_user(admin_id)
                if admin["role"] not in ["ADMIN", "SUPER_ADMIN"]:
                    raise UserAuthorizationError(admin_id, "deactivate_user", "Insufficient role")
            
            if not user["active"]:
                raise UserValidationError("User already inactive", "active", False, user_id=user_id)
            
            user["active"] = False
            logger.info(f"User {user_id} deactivated by {admin_id or 'system'}")
            return True
        
        except (UserNotFoundError, UserValidationError, UserAuthorizationError) as e:
            logger.error(f"Deactivation failed: {e}")
            raise
        except Exception as e:
            logger.error(f"Unexpected error deactivating user {user_id}: {e}")
            raise UserError("Failed to deactivate user", code=500, user_id=user_id) from e
    
    def update_email(self, user_id, new_email):
        try:
            user = self.get_user(user_id)
            
            if not new_email or "@" not in new_email:
                raise UserValidationError("Invalid email format", "email", new_email, user_id=user_id)
            
            if any(u["email"] == new_email for uid, u in self.users.items() if uid != user_id):
                raise UserEmailError(new_email, "Email already registered", user_id=user_id)
            
            old_email = user["email"]
            user["email"] = new_email
            logger.info(f"User {user_id} email changed from {old_email} to {new_email}")
            return True
        
        except (UserNotFoundError, UserValidationError, UserEmailError) as e:
            logger.error(f"Email update failed: {e}")
            raise
        except Exception as e:
            logger.error(f"Unexpected error updating email for user {user_id}: {e}")
            raise UserError("Failed to update email", code=500, user_id=user_id) from e
    
    def list_users(self, role=None):
        try:
            if role:
                if role not in ["USER", "ADMIN", "SUPER_ADMIN"]:
                    raise UserValidationError("Invalid role filter", "role", role)
                return [u for u in self.users.values() if u["role"] == role]
            return list(self.users.values())
        
        except UserValidationError as e:
            logger.error(f"List users error: {e}")
            raise
        except Exception as e:
            logger.error(f"Unexpected error listing users: {e}")
            raise UserError("Failed to list users", code=500) from e

import logging
from datetime import datetime, timedelta
from enum import Enum

logger = logging.getLogger(__name__)

class TaskError(Exception):
    def __init__(self, message, code=500, task_id=None, details=None):
        self.message = message
        self.code = code
        self.task_id = task_id
        self.details = details or {}
        super().__init__(message)

class TaskNotFoundError(TaskError):
    def __init__(self, task_id):
        super().__init__(f"Task {task_id} not found", code=404, task_id=task_id)

class TaskValidationError(TaskError):
    def __init__(self, message, field, value, task_id=None):
        super().__init__(message, code=400, task_id=task_id)
        self.field = field
        self.value = value

class TaskStatusError(TaskError):
    def __init__(self, task_id, current_status, attempted_status, allowed):
        super().__init__(
            f"Cannot transition from {current_status} to {attempted_status}. Allowed: {allowed}",
            code=400,
            task_id=task_id
        )
        self.current_status = current_status
        self.attempted_status = attempted_status
        self.allowed = allowed

class TaskDeadlineError(TaskError):
    def __init__(self, task_id, deadline, days_until):
        super().__init__(
            f"Task {task_id} deadline {deadline} is in {days_until} days",
            code=409,
            task_id=task_id
        )
        self.deadline = deadline
        self.days_until = days_until

class TaskAssignmentError(TaskError):
    def __init__(self, task_id, assignee, reason):
        super().__init__(
            f"Cannot assign task {task_id} to {assignee}: {reason}",
            code=400,
            task_id=task_id
        )
        self.assignee = assignee
        self.reason = reason

class TaskSystem:
    def __init__(self):
        self.tasks = {}
        self.next_id = 1
        self.valid_statuses = ["NEW", "IN_PROGRESS", "REVIEW", "COMPLETED", "CANCELLED"]
        self.transitions = {
            "NEW": ["IN_PROGRESS", "CANCELLED"],
            "IN_PROGRESS": ["REVIEW", "CANCELLED"],
            "REVIEW": ["COMPLETED", "IN_PROGRESS"],
            "COMPLETED": [],
            "CANCELLED": []
        }
    
    def create_task(self, title, description, assigned_to=None, priority="MEDIUM", deadline=None):
        try:
            if not title or len(title) < 3:
                raise TaskValidationError("Title must be at least 3 characters", "title", title)
            
            if description is None:
                raise TaskValidationError("Description is required", "description", description)
            
            if priority not in ["LOW", "MEDIUM", "HIGH", "CRITICAL"]:
                raise TaskValidationError("Invalid priority", "priority", priority)
            
            if deadline:
                try:
                    deadline_date = datetime.fromisoformat(deadline)
                    if deadline_date < datetime.now():
                        raise TaskValidationError("Deadline must be in the future", "deadline", deadline)
                    days_until = (deadline_date - datetime.now()).days

                raise EventValidationError("Max participants must be positive", "max_participants", max_participants)

                "id": event_id,
                "name": name,
                "date": date,
                "location": location,
                "max_participants": max_participants,
                "registered": 0,
                "status": "ACTIVE"

    def __init__(self, document_id, version, reason, details=None):

            code=409,
            document_id=document_id,
            details=details
        )
        self.version = version
        self.reason = reason

class DocumentSystem:
    def __init__(self):
        self.documents = {}
        self.next_id = 1
        self.allowed_types = [".txt", ".pdf", ".docx", ".jpg", ".png", ".json"]
        self.max_size = 10 * 1024 * 1024  # 10 MB
    
    def upload_document(self, name, content, file_type, metadata=None):
        try:
            if not name or len(name) < 3:
                raise DocumentValidationError("Name must be at least 3 characters", "name", name)
            
            if not content:
                raise DocumentValidationError("Content cannot be empty", "content", None)
            
            file_type_lower = file_type.lower()
            if file_type_lower not in self.allowed_types:
                raise DocumentTypeError(name, file_type, self.allowed_types)
            
            size = len(content)
            if size > self.max_size:
                raise DocumentSizeError(name, size, self.max_size)
            
            # Проверка на дубликаты по хешу
            content_hash = hashlib.md5(content).hexdigest()
            for doc in self.documents.values():
                if doc.get("hash") == content_hash:
                    raise DocumentValidationError(
                        "Document already exists (duplicate content)",
                        "hash",
                        content_hash[:8]
                    )
            
            document_id = self.next_id
            self.documents[document_id] = {
                "id": document_id,
                "name": name,
                "type": file_type,
                "size": size,
                "hash": content_hash,
                "metadata": metadata or {},
                "version": 1,
                "created_at": datetime.now(),
                "updated_at": datetime.now()
            }
            self.next_id += 1
            logger.info(f"Document {document_id} uploaded: {name} ({size} bytes)")
            return document_id
        
        except (DocumentValidationError, DocumentTypeError, DocumentSizeError) as e:
            logger.error(f"Upload failed: {e}")
            raise
        except Exception as e:
            logger.error(f"Unexpected error uploading document: {e}")
            raise DocumentError("Failed to upload document", code=500) from e

        doc = self.documents.get(document_id)
        if not doc:
            raise DocumentNotFoundError(document_id)

                raise DocumentValidationError("Content cannot be empty", "content", None, document_id=document_id)
            
            new_size = len(new_content)
            if new_size > self.max_size:
                raise DocumentSizeError(doc["name"], new_size, self.max_size)
            
            new_hash = hashlib.md5(new_content).hexdigest()
            if new_hash == doc["hash"]:
                raise DocumentVersionError(document_id, doc["version"], "No changes detected")
            
            doc["version"] += 1

    def get_session(self, session_id):
        session = self.sessions.get(session_id)
        if not session:
            raise SessionNotFoundError(session_id)
        
        if session["status"] != "ACTIVE":
            raise SessionExpiredError(session_id, session.get("expired_at"), 0)
        
        # Проверка истечения по неактивности
        last_activity = session["last_activity"]
        if datetime.now() - last_activity > timedelta(seconds=self.session_timeout):
            session["status"] = "EXPIRED"
            session["expired_at"] = datetime.now()
            days_inactive = (datetime.now() - last_activity).days
            raise SessionExpiredError(session_id, session["expired_at"], days_inactive)
        
        return session

        try:
            session = self.get_session(session_id)
            session["last_activity"] = datetime.now()
            logger.info(f"Session {session_id} refreshed")

        except (SessionNotFoundError, SessionExpiredError) as e:

# inventory_system.py
import logging
from datetime import datetime
import random

logger = logging.getLogger(__name__)

class InventoryError(Exception):
    def __init__(self, message, code=500, item_id=None, details=None):
        self.message = message
        self.code = code
        self.item_id = item_id
        self.details = details or {}
        super().__init__(message)

class ItemNotFoundError(InventoryError):
    def __init__(self, item_id):
        super().__init__(f"Item {item_id} not found", code=404, item_id=item_id)

class ItemValidationError(InventoryError):
    def __init__(self, message, field, value, item_id=None):
        super().__init__(message, code=400, item_id=item_id)
        self.field = field
        self.value = value

class ItemQuantityError(InventoryError):
    def __init__(self, item_id, requested, available, reason, details=None):
        super().__init__(
            f"Item {item_id} quantity error: requested {requested}, available {available}. {reason}",
            code=409,
            item_id=item_id,
            details=details
        )
        self.requested = requested
        self.available = available
        self.reason = reason

class ItemLocationError(InventoryError):
    def __init__(self, item_id, current_location, attempted_location, reason, details=None):
        super().__init__(
            f"Item {item_id} location error: cannot move from {current_location} to {attempted_location}. {reason}",
            code=400,
            item_id=item_id,
            details=details
        )
        self.current_location = current_location
        self.attempted_location = attempted_location
        self.reason = reason

class ItemExpiryError(InventoryError):
    def __init__(self, item_id, expiry_date, days_until):
        super().__init__(
            f"Item {item_id} expires on {expiry_date} (in {days_until} days)",
            code=409,
            item_id=item_id
        )
        self.expiry_date = expiry_date
        self.days_until = days_until

class InventorySystem:
    def __init__(self):
        self.items = {}
        self.next_id = 1
        self.locations = ["A1", "A2", "B1", "B2", "C1", "C2"]
    
    def add_item(self, name, quantity, location, expiry_date=None, category=None):
        try:
            if not name or len(name) < 2:
                raise ItemValidationError("Name must be at least 2 characters", "name", name)
            
            if quantity <= 0:
                raise ItemValidationError("Quantity must be positive", "quantity", quantity)
            
            if location not in self.locations:
                raise ItemValidationError("Invalid location", "location", location)
            
            # Проверка истечения срока
            if expiry_date:
                try:
                    expiry = datetime.fromisoformat(expiry_date)
                    if expiry < datetime.now():
                        raise ItemValidationError("Expiry date must be in the future", "expiry_date", expiry_date)
                    days_until = (expiry - datetime.now()).days
                    if days_until < 7:
                        raise ItemExpiryError(None, expiry_date, days_until)
                except ValueError:
                    raise ItemValidationError("Invalid expiry date format", "expiry_date", expiry_date)
            
            item_id = self.next_id
            self.items[item_id] = {
                "id": item_id,
                "name": name,
                "quantity": quantity,
                "location": location,
                "expiry_date": expiry_date,
                "category": category,
                "status": "ACTIVE",
                "created_at": datetime.now(),
                "updated_at": datetime.now()
            }
            self.next_id += 1
            logger.info(f"Item {item_id} added: {name} x{quantity} at {location}")
            return item_id
        
        except (ItemValidationError, ItemExpiryError) as e:
            logger.error(f"Add item failed: {e}")
            raise
        except Exception as e:
            logger.error(f"Unexpected error adding item: {e}")
            raise InventoryError("Failed to add item", code=500) from e
    
    def get_item(self, item_id):
        item = self.items.get(item_id)
        if not item:
            raise ItemNotFoundError(item_id)
        
        # Проверка истёкших товаров
        if item["expiry_date"] and item["status"] == "ACTIVE":
            expiry = datetime.fromisoformat(item["expiry_date"])
            if expiry < datetime.now():
                item["status"] = "EXPIRED"
                days_overdue = (datetime.now() - expiry).days
                raise ItemExpiryError(item_id, item["expiry_date"], -days_overdue)
        
        return item
    
    def move_item(self, item_id, new_location, quantity):
        try:
            item = self.get_item(item_id)
            
            if item["status"] != "ACTIVE":
                raise ItemQuantityError(item_id, quantity, item["quantity"], f"Item is {item['status']}")
            
            if new_location not in self.locations:
                raise ItemValidationError("Invalid location", "location", new_location, item_id=item_id)
            
            if quantity <= 0:
                raise ItemValidationError("Quantity must be positive", "quantity", quantity, item_id=item_id)
            
            if quantity > item["quantity"]:
                raise ItemQuantityError(item_id, quantity, item["quantity"], "Insufficient quantity")
            
            if new_location == item["location"]:
                raise ItemLocationError(item_id, item["location"], new_location, "Cannot move to same location")
            
            old_location = item["location"]
            item["location"] = new_location
            item["updated_at"] = datetime.now()
            logger.info(f"Item {item_id} moved from {old_location} to {new_location}")
            return True
        
        except (ItemNotFoundError, ItemValidationError, ItemQuantityError, ItemLocationError, ItemExpiryError) as e:
            logger.error(f"Move failed: {e}")
            raise
        except Exception as e:
            logger.error(f"Unexpected error moving item {item_id}: {e}")
            raise InventoryError("Failed to move item", code=500, item_id=item_id) from e
    
    def adjust_quantity(self, item_id, delta):
        try:
            item = self.get_item(item_id)
            
            if item["status"] != "ACTIVE":
                raise ItemQuantityError(item_id, delta, item["quantity"], f"Item is {item['status']}")
            
            new_quantity = item["quantity"] + delta
            if new_quantity < 0:
                raise ItemQuantityError(item_id, abs(delta), item["quantity"], "Quantity would become negative")
            
            if delta == 0:
                raise ItemValidationError("No change", "delta", 0, item_id=item_id)
            
            item["quantity"] = new_quantity
            item["updated_at"] = datetime.now()
            logger.info(f"Item {item_id} quantity adjusted by {delta} to {new_quantity}")
            return new_quantity
        
        except (ItemNotFoundError, ItemQuantityError, ItemValidationError, ItemExpiryError) as e:
            logger.error(f"Quantity adjustment failed: {e}")
            raise
        except Exception as e:
            logger.error(f"Unexpected error adjusting quantity for item {item_id}: {e}")
            raise InventoryError("Failed to adjust quantity", code=500, item_id=item_id) from e
    
    def get_items_by_location(self, location):
        try:
            if location not in self.locations:
                raise ItemValidationError("Invalid location", "location", location)
            return [i for i in self.items.values() if i["location"] == location]
        
        except ItemValidationError as e:
            raise
        except Exception as e:
            logger.error(f"Unexpected error getting items by location: {e}")
            raise InventoryError("Failed to get items", code=500) from e
    
    def search_items(self, query, category=None):
        try:
            results = []
            for item in self.items.values():
                if item["status"] != "ACTIVE":
                    continue
                if query and query.lower() not in item["name"].lower():
                    continue
                if category and item["category"] != category:
                    continue
                results.append(item)
            return results
        
        except Exception as e:
            logger.error(f"Unexpected error searching items: {e}")
            raise InventoryError("Failed to search items", code=500) from e
