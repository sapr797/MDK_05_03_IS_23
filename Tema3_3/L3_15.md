## 1. Введение: Почему ошибки доступа к файлам — это отдельная категория

В реальных приложениях файловый ввод-вывод редко проходит гладко. Пользователи могут:

- Удалить файл, который программа ожидает найти.
- Изменить права доступа к файлу или директории.
- Заполнить диск до отказа.
- Работать на сетевом диске с задержками.

Эти ситуации приводят к специфическим исключениям, которые **должны быть протестированы** так же тщательно, как и основной функционал.

**Основные ошибки доступа к файлам в Python:**

| Исключение | Когда возникает |
|------------|-----------------|
| `FileNotFoundError` | Файл или директория не существует |
| `PermissionError` | Недостаточно прав для чтения/записи/удаления |
| `IsADirectoryError` | Ожидался файл, но это директория |
| `NotADirectoryError` | Ожидалась директория, но это файл |
| `OSError` | Общая ошибка ОС (заполненный диск, сетевой сбой и др.) |
| `FileExistsError` | Файл уже существует при попытке создания |

3. Тестирование PermissionError
3.1. Реализация кода
python
import os
import stat
from pathlib import Path

class FileManager:
    def read_file(self, file_path):
        path = Path(file_path)
        if not path.exists():
            raise FileNotFoundError(f"File not found: {file_path}")
        
        try:
            with open(path, 'r') as f:
                return f.read()
        except PermissionError:
            raise PermissionError(f"No read permission for: {file_path}")
    
    def write_file(self, file_path, content):
        path = Path(file_path)
        try:
            with open(path, 'w') as f:
                f.write(content)
        except PermissionError:
            raise PermissionError(f"No write permission for: {file_path}")
    
    def delete_file(self, file_path):
        path = Path(file_path)
        if not path.exists():
            raise FileNotFoundError(f"File not found: {file_path}")
        
        try:
            path.unlink()
        except PermissionError:
            raise PermissionError(f"No delete permission for: {file_path}")
    
    def create_directory(self, dir_path):
        path = Path(dir_path)
        try:
            path.mkdir(parents=True, exist_ok=True)
        except PermissionError:
            raise PermissionError(f"Cannot create directory: {dir_path}")

import pytest
import os
import stat
from pathlib import Path

def test_write_file_permission_error(tmp_path):
    manager = FileManager()
    
    # Создаём файл и делаем его доступным только для чтения
    file_path = tmp_path / "readonly.txt"
    file_path.touch()
    # chmod 444: только чтение для всех
    os.chmod(file_path, stat.S_IRUSR | stat.S_IRGRP | stat.S_IROTH)
    
    with pytest.raises(PermissionError) as exc_info:
        manager.write_file(file_path, "New content")
    
    assert "No write permission for:" in str(exc_info.value)
    assert str(file_path) in str(exc_info.value)

def test_read_file_permission_error(tmp_path):

    # Реальный сценарий: файл с правами 000
    file_path = tmp_path / "no_perm.txt"
    file_path.write_text("Secret content")
    os.chmod(file_path, 0)  # Никаких прав
    
    with pytest.raises(PermissionError) as exc_info:
        manager.read_file(file_path)
    
    assert "No read permission for:" in str(exc_info.value)

        manager.delete_file(file_path)
    
    # Восстанавливаем права для очистки (pytest автоматически удалит tmp_path)

def test_write_file_permission_error_with_monkeypatch(monkeypatch):
    manager = FileManager()
    
    # Счётчик вызовов для подмены только при записи
    def mock_open_write(*args, **kwargs):
        if 'w' in args or kwargs.get('mode') == 'w':
            raise PermissionError("Mocked permission error")
        return open(*args, **kwargs)
    
    monkeypatch.setattr('builtins.open', mock_open_write)
    file_path = Path("/tmp/test.txt")
    
    with pytest.raises(PermissionError) as exc_info:
        manager.write_file(file_path, "Content")
    
    assert "No write permission for:" in str(exc_info.value)

class LogHandler:
    def append_log(self, file_path, message):
        path = Path(file_path)
        try:
            with open(path, 'a') as f:
                f.write(f"{message}\n")
        except FileNotFoundError:
            # Пытаемся создать родительскую директорию
            try:
                path.parent.mkdir(parents=True, exist_ok=True)
                # Повторная попытка
                with open(path, 'a') as f:
                    f.write(f"{message}\n")
            except PermissionError:
                raise PermissionError(f"Cannot create log file: {file_path}")
        except PermissionError:
            raise PermissionError(f"No write permission for log: {file_path}")

class BackupSystem:
    def __init__(self, source_dir, backup_dir):
        self.source_dir = Path(source_dir)
        self.backup_dir = Path(backup_dir)
    
    def create_backup(self, filename):
        source = self.source_dir / filename
        if not source.exists():
            raise FileNotFoundError(f"Source file not found: {filename}")
        
        if not source.is_file():
            raise IsADirectoryError(f"Source is a directory: {filename}")
        
        try:
            self.backup_dir.mkdir(parents=True, exist_ok=True)
        except PermissionError:
            raise PermissionError(f"Cannot create backup directory: {self.backup_dir}")
        
        target = self.backup_dir / filename
        try:
            import shutil
            shutil.copy2(source, target)
        except PermissionError:
            raise PermissionError(f"Cannot write to backup directory: {self.backup_dir}")
        
        return target

def test_backup_system_success(tmp_path):
    source_dir = tmp_path / "source"
    backup_dir = tmp_path / "backup"
    source_dir.mkdir()
    
    system = BackupSystem(source_dir, backup_dir)
    
    # Создаём исходный файл
    source_file = source_dir / "data.txt"
    source_file.write_text("Backup me")
    
    result = system.create_backup("data.txt")

9. Итоговый вывод по лекции
Тестирование ошибок доступа к файлам — это важная часть обеспечения надёжности приложения. Пользователи могут столкнуться с самыми разными ситуациями, и ваша программа должна корректно обрабатывать каждую из них.

Ключевые тезисы для запоминания:

FileNotFoundError — тестируйте все сценарии, где файл может отсутствовать.

PermissionError — тестируйте с помощью os.chmod для ограничения прав.

IsADirectoryError и NotADirectoryError — проверяйте при работе с файлами и директориями.

FileExistsError — проверяйте при создании файлов без exist_ok.

Используйте tmp_path для безопасного создания файлов в тестах.

Используйте monkeypatch для симуляции ошибок в системных вызовах.

Всегда восстанавливайте права доступа (или используйте tmp_path для автоматической очистки).

import os
import json
from pathlib import Path

class FileProcessor:
    def __init__(self, base_dir):
        self.base_dir = Path(base_dir)
    
    def read_json(self, filename):
        file_path = self.base_dir / filename
        if not file_path.exists():
            raise FileNotFoundError(f"File not found: {filename}")
        if not file_path.is_file():
            raise IsADirectoryError(f"Path is a directory: {filename}")
        
        try:
            with open(file_path, 'r') as f:
                return json.load(f)
        except PermissionError:
            raise PermissionError(f"Cannot read file: {filename}")
    
    def write_json(self, filename, data):
        file_path = self.base_dir / filename
        if file_path.exists() and file_path.is_dir():
            raise IsADirectoryError(f"Cannot write to directory: {filename}")
        
        try:
            file_path.parent.mkdir(parents=True, exist_ok=True)
            with open(file_path, 'w') as f:
                json.dump(data, f)
        except PermissionError:
            raise PermissionError(f"Cannot write to file: {filename}")
    
    def delete_json(self, filename):
        file_path = self.base_dir / filename
        if not file_path.exists():
            raise FileNotFoundError(f"File not found: {filename}")
        if not file_path.is_file():
            raise IsADirectoryError(f"Path is a directory: {filename}")
        
        try:
            file_path.unlink()
        except PermissionError:
            raise PermissionError(f"Cannot delete file: {filename}")
    
    def read_all_json(self):
        try:
            files = list(self.base_dir.glob("*.json"))
            result = {}
            for file_path in files:
                result[file_path.stem] = self.read_json(file_path.name)
            return result
        except PermissionError:
            raise PermissionError(f"Cannot read directory: {self.base_dir}")
